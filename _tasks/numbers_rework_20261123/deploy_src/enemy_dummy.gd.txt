class_name EnemyDummy
extends Node3D
## 木人桩：受击闪白 + 弹簧晃动 + 伤害数字；击破后延时复活。
## kind: 0 地面  1 浮空  2 绕圈巡游

const CS := preload("res://scripts/combat_stats.gd")      # 2026-11-23 飘字换成大世界的口径（只改显示）
const Profile := preload("res://scripts/raid_profile.gd")

var kind := 0
var max_hp := 400.0
var hp := 400.0
var alive := true
var home := Vector3.ZERO

var _t := 0.0
var _wob := Vector3.ZERO
var _wob_v := Vector3.ZERO
var _flash := 0.0
var _model: Node3D
var _mat: StandardMaterial3D
var _bar: MeshInstance3D
var _respawn := 0.0
## 空战拳脚的击退（air_melee.gd）：浮空 / 巡游的木桩会被打退、击飞，撞地砸坑，随后自己飘回原位
var _kv := Vector3.ZERO
var _koff := Vector3.ZERO
var _kt := 99.0


func _ready() -> void:
	add_to_group("enemies")
	_t = randf() * 10.0
	home = position
	_model = Node3D.new()
	add_child(_model)
	_mat = MeshLib.std_mat(Color(0.78, 0.33, 0.27), 0.75)
	_mat.emission_enabled = true
	_mat.emission = Color(1, 1, 1)
	_mat.emission_energy_multiplier = 0.0
	var wood := MeshLib.std_mat(Color(0.42, 0.4, 0.4), 0.9)

	var torso := CapsuleMesh.new()
	torso.radius = 0.32
	torso.height = 1.0
	_add(torso, _mat, Vector3(0, 1.1, 0))
	var head := SphereMesh.new()
	head.radius = 0.24
	head.height = 0.48
	_add(head, _mat, Vector3(0, 1.88, 0))
	var arm := CylinderMesh.new()
	arm.top_radius = 0.06
	arm.bottom_radius = 0.06
	arm.height = 1.3
	_add(arm, wood, Vector3(0, 1.3, 0), Vector3(0, 0, 90))
	if kind == 0:
		var post := CylinderMesh.new()
		post.top_radius = 0.1
		post.bottom_radius = 0.14
		post.height = 0.7
		_add(post, wood, Vector3(0, 0.35, 0))
		var base := CylinderMesh.new()
		base.top_radius = 0.4
		base.bottom_radius = 0.45
		base.height = 0.1
		_add(base, wood, Vector3(0, 0.05, 0))
	else:
		var ring := TorusMesh.new()
		ring.inner_radius = 0.38
		ring.outer_radius = 0.46
		_add(ring, wood, Vector3(0, 0.45, 0))

	_bar = MeshInstance3D.new()
	var qm := QuadMesh.new()
	qm.size = Vector2(1.0, 0.09)
	_bar.mesh = qm
	var bm := StandardMaterial3D.new()
	bm.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED
	bm.albedo_color = Color(0.95, 0.3, 0.25)
	bm.billboard_mode = BaseMaterial3D.BILLBOARD_ENABLED
	bm.billboard_keep_scale = true
	bm.no_depth_test = true
	_bar.material_override = bm
	_bar.position.y = 2.4
	_bar.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
	add_child(_bar)


func _add(mesh: Mesh, mat: Material, pos: Vector3, rot_deg := Vector3.ZERO) -> void:
	var mi := MeshInstance3D.new()
	mi.mesh = mesh
	mi.material_override = mat
	mi.position = pos
	mi.rotation_degrees = rot_deg
	_model.add_child(mi)


func aim_point() -> Vector3:
	return global_position + Vector3(0.0, 1.15, 0.0)


func is_launched() -> bool:
	return alive and kind != 0 and _kv.length() > 9.0


func velocity() -> Vector3:
	return _kv


## 拳脚 / 重击：扣血 + 击退（地面桩只晃不动）。返回 0 命中 / -1 无效
func air_hit(_attacker: Node3D, info: Dictionary) -> int:
	if not alive:
		return -1
	var dir: Vector3 = info.get("dir", Vector3.FORWARD)
	take_hit(float(info.get("dmg", 10.0)), dir, Color(1.0, 0.9, 0.7))
	if kind != 0 and alive:
		_kv = dir * float(info.get("kb", 3.0))
		_kt = 0.0
	return 0


## 2026-11-23 数值重构：主城（木人 / 刀客 / 剑冢首领）扣血照旧按原始伤害（base(境界) × 倍率、对手血也按 base 配），
## 飘字只在显示时换成和大世界一个口径：招式单位（÷ 主角境界的 base）× 主角同级打同级怪一个单位的伤害（CS.unit_same）。
## 不然大乘在大世界里看到「1.4亿」，回试剑台只看到「110」。主角的等级取存档里的修为（Profile.cult，按主角现在的境界对齐）
static func shown_dmg(n: Node, dmg: float) -> float:
	var realm := Profile.realm()
	if n != null and n.is_inside_tree():
		var p := n.get_tree().get_first_node_in_group("player")
		if p != null:
			realm = int(p.get("realm_idx"))
	var lv := int(Profile.cult(realm)[0])
	return dmg / CS.base(realm) * CS.unit_same(lv)


func take_hit(dmg: float, dir: Vector3, color: Color) -> void:
	if not alive:
		return
	hp -= dmg
	_flash = 1.0
	_wob_v += Vector3(dir.x, 0.0, dir.z) * 7.0
	Vfx.damage_number(aim_point() + Vector3(0, 0.8, 0), shown_dmg(self, dmg), color)
	if hp <= 0.0:
		alive = false
		_respawn = 2.5
		# 碎成方块再消融：按各部件的基本体体积采样成小方块（颜色取部件材质）
		var parts := []
		for n in _model.get_children():
			var mi := n as MeshInstance3D
			parts.append([mi, (mi.material_override as StandardMaterial3D).albedo_color])
		var hit := aim_point() - dir.normalized() * 0.25
		Vfx.shatter(_model, hit, dir, Color.WHITE.lerp(color, 0.55), 1.1, parts, 0.11)
		_model.visible = false
		_bar.visible = false
		Vfx.ribbon_burst(hit, Vector3.UP, color, 8, 7.0, 160.0, 0.4)


func _process(dt: float) -> void:
	_t += dt
	if kind != 0:
		_tick_knock(dt)
	if kind == 1:
		position = home + Vector3(0, sin(_t * 0.9) * 1.2, 0) + _koff
	elif kind == 2:
		position = home + Vector3(cos(_t * 0.55) * 7.0, sin(_t * 0.8) * 0.8, sin(_t * 0.55) * 7.0) + _koff

	if not alive:
		_respawn -= dt
		if _respawn <= 0.0:
			alive = true
			hp = max_hp
			_model.visible = true
			_bar.visible = true
			_model.scale = Vector3.ONE * 0.01
			var tw := create_tween()
			tw.tween_property(_model, "scale", Vector3.ONE, 0.3).set_trans(Tween.TRANS_BACK).set_ease(Tween.EASE_OUT)
		return

	_flash = maxf(_flash - dt * 7.0, 0.0)
	_mat.emission_energy_multiplier = _flash * 1.2
	_wob_v += (-_wob * 140.0 - _wob_v * 9.0) * dt
	_wob += _wob_v * dt
	_model.rotation.x = clampf(_wob.z * 0.16, -0.6, 0.6)
	_model.rotation.z = clampf(-_wob.x * 0.16, -0.6, 0.6)
	_bar.scale.x = maxf(hp / max_hp, 0.001)


## 击退位移：飞出去 → 减速 → 一段时间没再挨打就弹回原位；飞行途中砸到地面则砸坑停下
func _tick_knock(dt: float) -> void:
	const Combat := preload("res://scripts/combat.gd")
	var cdt: float = Combat.cdt(dt)
	_kt += cdt
	if _kv.length_squared() > 0.01:
		var spd := _kv.length()
		_koff += _kv * cdt
		_kv -= _kv * minf((0.7 if spd > 9.0 else 5.0) * cdt, 1.0)
		if spd > 9.0 and position.y <= 0.4 and _kv.y < 0.0:
			_koff.y += 0.4 - position.y
			if alive:
				var p := Vector3(global_position.x, 0.0, global_position.z)
				Vfx.dun_slam(p, Color(1.0, 0.9, 0.75), clampf(spd / 34.0, 0.7, 1.5))
				Vfx.crater(p, clampf(spd / 34.0, 0.7, 1.5))
				take_hit(40.0 * spd / 34.0, Vector3.DOWN, Color(1.0, 0.9, 0.7))
				for pl in get_tree().get_nodes_in_group("player"):
					(pl as Player).air.on_crash(self, p, spd, true)
			_kv = Vector3.ZERO
	if _kt > 1.6 or not alive:
		_koff = _koff.lerp(Vector3.ZERO, 1.0 - exp(-2.5 * dt))
