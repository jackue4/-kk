## from = 怪倒下的地方（Vector3.INF = 不放光点、直接记到显示的数上；给了就从那里飞心得光点，见 insight_fx.gd）；kind：0 普通 / 1 精英 / 2 首领、修士
## 2026-11-23：ri = 怪所在的区域（必填：同一个等级在不同区域可能是不同的境界档，心得的修为单位跟着档走）
func add_insight(mob_lv: int, k: float, from: Vector3, kind: int, ri: int) -> float:
	if not running:
		return 0.0      # 已经结算 / 断线了：这一局的心得不再记，也不放光点
	var cap := insight_cap()
	var got := minf(kill_exp(mob_lv, k, ri), maxf(cap - insight, 0.0))
	insight += got
	if got > 0.0:
		if from.is_finite() and _ins_fx != null and is_instance_valid(_ins_fx):
			_ins_fx.mobile = mobile
			_ins_fx.launch(from, got, kind)
		else:
			insight_arrived(got)
	if insight >= cap - 0.01:
		if not _insight_full:
			_insight_full = true
			hud.toast("心得已满（本局上限）", Color(0.95, 0.8, 0.45))
	else:
		_insight_full = false
	return got


## 心得光点飞到身上了（insight_fx.gd 调）：信息牌上的数加上这一份（不超过真正记上的）

	elif insight_shown < insight and (_ins_fx == null or not is_instance_valid(_ins_fx) or int(_ins_fx.pending()) == 0):

func _count_keys(list: Array, to: int) -> int:

func _trim_keys(list: Array, keep: int, to: int) -> void:
	var n := _count_keys(list, to)
	for i in range(list.size() - 1, -1, -1):
		if n <= keep:
			return
		var it: Variant = list[i]
		if it is Dictionary and str((it as Dictionary).get("kind", "")) == "key" and int((it as Dictionary).get("to", 0)) == to:
			list.remove_at(i)
			n -= 1

		var so: Object = null

			# 以前只扣到 0、不置 dead，靠每帧 hp <= 0 兜底结算，结算那几秒人还能飞来飞去

## PvP 封顶：一下最多 CS.PVP_HIT_CAP；同一个人连着打，先有 PVP_BURST 的余量，之后按每秒 PVP_SUSTAIN 慢慢回（境界差再大，满血打死也要三秒以上）
var _pvp_heat := {}      # peer id -> [上次结算的 run_t, 已经用掉的量]


## 2026-11-23：k = CS.pvp_cap_k(对方境界, 我的境界)，三道封顶（单下 / 余量 / 每秒）一起乘
func _pvp_clamp(from: int, d: float, k := 1.0) -> float:
	var h: Array = _pvp_heat.get(from, [run_t, 0.0])
	var used := maxf(float(h[1]) - (run_t - float(h[0])) * CS.PVP_SUSTAIN * k, 0.0)
	var v := minf(minf(d, CS.PVP_HIT_CAP * k), maxf(CS.PVP_BURST * k - used, 0.0))
	_pvp_heat[from] = [run_t, used + v]
	return v

func _on_trib_done(ok: bool, perfect: int, total: int) -> void:
	var to := _trib_to
	_trib_to = -1
	_trib_free = false
	if ok:
		var r0 := player.realm_idx
		var pw0 := CS.power(r0, lv, fx)
		## 2026-11-23 照 Hammer 结转：瓶颈那一级之外多攒的修为按 carry_k 换成新境界的单位带过去（GM 直接渡劫、没到瓶颈的是 0）
		var carry := Prog.carry_over(lv, xp, r0)
		xp = 0.0
		lv = maxi(lv + 1, to * Data.LV_PER_REALM + 1)   # GM 直接渡劫时等级跟到新境界的起点
		player.set_realm(to)     # 自带升阶特效和右侧的毛笔题字

		var bonus := Prog.xp_need(lv) * 0.3 * float(perfect) / maxf(float(total), 1.0)
		_trib_lv0 = lv
		var res := Prog.add_xp(lv, 0.0, player.realm_idx, carry + bonus)
		lv = int(res[0])
		xp = float(res[1])

		## 60 秒内不自动再起劫（_trib_hold），到点了还在瓶颈就照常再起
		xp = maxf(xp - Prog.xp_need(lv) * 0.5, 0.0)

		player.melee.hp = maxf(player.melee.hp, 30.0)

func _beam(parent: Node3D, col: Color, height: float, width: float) -> Node3D:

## 全房间共用的时间（联机 = 房主建房起算的 net.room_t；单人 = 本局自己的）。渡劫时也照走
func world_t() -> float:
	return float(net.room_t) if net_on() else _solo_t


func _room_seed() -> int:
	return int(net.seed_v) if net_on() else _solo_seed


## 离下一次传送阵轮换还有几秒
func rotation_left() -> float:
	return ROT_PERIOD - fmod(world_t(), ROT_PERIOD)


## 按房间时间算这是第几轮，轮次变了就重排：用（种子, 轮次）洗牌，前两座是古传送阵、第三座是灵石阵、其余熄着。
## 谁在什么时候算出来都一样（中途加入的人也一样），不用发消息
func _rotate_gates(force := false) -> void:
	var k := int(floorf(world_t() / ROT_PERIOD))
	if k == _rot_k and not force:
		return
	_rot_k = k
	var gates: Array = []
	for e: Dictionary in _extracts:
		if e.has("base"):
			gates.append(e)
	if gates.is_empty():
		return
	var order := range(gates.size())
	var srng := RandomNumberGenerator.new()
	srng.seed = _room_seed() + 7919 * k
	for i in range(order.size() - 1, 0, -1):
		var j := srng.randi_range(0, i)
		var tmp: int = order[i]
		order[i] = order[j]
		order[j] = tmp
	var lit: Array = []
	var free_n := int(geo.gates_free())      # 大世界 2；新图按 map_def（沙漠 1、研究所 2），再加一座灵石阵
	for i in gates.size():
		var e: Dictionary = gates[i]
		var o := order.find(i)
		var kind := "paid" if o == free_n else "gate"
		var on: bool = o < free_n + 1
		var base := str(e["base"])
		var nm := base if kind == "gate" else "%s灵石阵" % base.trim_suffix("传送阵")
		if bool(e["active"]) != on or str(e["kind"]) != kind:
			# 2026-11-20 修：正站在这座阵里读条的，阵换了种类 / 熄了就从头读（免费阵读到一半变成灵石阵，读满时会悄悄扣灵石）；
			# 探明过的阵变成灵石阵、人又在附近：再报一次要献多少灵石（探明时只报一次）
			if not force and player != null:
				var pp := player.global_position
				var ep: Vector3 = e["pos"]
				var dd := Vector2(pp.x - ep.x, pp.z - ep.z).length()
				if dd < EXTRACT_R and absf(pp.y - ep.y) < 8.0:
					extract_k = 0.0
				if on and kind == "paid" and bool(e["known"]) and dd < 170.0 and hud != null:
					hud.toast("%s：献 %d 灵石（从洞府扣）即可撤离" % [nm, paid_cost()], Color(1.0, 0.85, 0.4))
			e["active"] = on
			e["kind"] = kind
			e["name"] = nm
			if is_instance_valid(e.get("node")):
				(e["node"] as Node3D).queue_free()
			e["node"] = _build_extract(e["pos"], on, EX_KINDS[kind]["col"])
		if on:
			lit.append(nm)
	if not force and hud != null:
		hud.toast("传送阵轮转：%s 亮了" % "、".join(lit), Color(0.4, 0.95, 1.0))

	st.begin(Mesh.PRIMITIVE_TRIANGLES)
	var fire := SurfaceTool.new()
	fire.begin(Mesh.PRIMITIVE_TRIANGLES)
	for i in 8:
		var a := TAU * float(i) / 8.0
		var o := Vector3(cos(a), 0.0, sin(a)) * 22.0
		var h := 9.0 if i % 2 == 0 else 5.5
		Data.add_box(st, Vector3(3.2, 1.0, 3.2), Transform3D(Basis(Vector3.UP, -a), o + Vector3(0, 0.5, 0)), Color(0.36, 0.35, 0.41))

func _spawn_mob(def_idx: int, ri: int, pos: Vector3, elite := false, lvl := -1, hp_k := 1.0) -> Node3D:
	var d: Dictionary = _mob_def(def_idx)
	var is_boss: bool = d.get("boss", false)
	if bool(d.get("elite_only", false)) and not is_boss:
		elite = true      # 新图的凝胶怪只以精英身份出现（ex_mobs 的 elite_only）
	var base: Array = Data.SHAPE_BASE[str(d["shape"])]
	var m := Mob.new()
	m.player = player
	m.def = d
	m.region = ri
	m.lv = lvl if lvl > 0 else _level_in(ri, pos)
	m.elite = elite
	var hp0 := (Data.BOSS_HP if is_boss else float(base[0])) * float(d["hp"])
	var at0 := (Data.BOSS_DMG if is_boss else float(base[1])) * float(d["atk"])
	## 2026-11-23 数值重构：血 / 攻换到新单位（HP_CONV / ATK_CONV：同级兽 4 下死、一下掉 8.7%，同以前）× 区域境界档的量级（CS.mob_scale，档内照旧按深度线性）；
	## 防御不存在怪身上——出手时按（区域, 等级）现算（_target_scale），木偶两边算出来一样。物种、精英、首领的倍数都在公式外面
	m.hp_max = hp0 * CS.HP_CONV * CS.mob_s_hp(ri, m.lv) * (Data.ELITE_HP if elite else 1.0) * hp_k
	m.atk = at0 * Data.MOB_DMG * CS.ATK_CONV * CS.mob_s_atk(ri, m.lv) * (Data.ELITE_DMG if elite else 1.0)
	m.xp_k = 14.0 if is_boss else float(base[3]) * (3.0 if elite else 1.0)
	if geo.fixed:
		## 新图（第 3 节）：难度倍率按（图, 层）乘在老公式外面；心得倍率在倒下时按位置乘（木偶那边同一个公式）；怪不穿墙
		var zk: Dictionary = geo.zone_k(pos)
		m.hp_max *= float(zk["hp"])
		m.atk *= float(zk["atk"])
		m.block_fn = Callable(level, "blocked")
		if not bool(d.get("loot", true)):
			m.xp_k = 0.0      # 首领召唤的小怪（幽冥泣）：不给心得、不掉东西
	m.dmg_in = player_dmg_vs.bind(m)
	m.atk_out = to_player_from.bind(m)
	m.floor_fn = _ground
	m.position = pos
	_nid += 1
	m.net_id = _nid
	m.set_meta("def_idx", def_idx)
	m.died.connect(_on_mob_died)
	_mob_root.add_child(m)
	return m

## 挪的那一下地标离主角有半张图（一千七百多米）远，看不见。
func _reimage(p: Vector3) -> void:
	for b: Dictionary in _bosses:
		var pos: Vector3 = b["pos"]
		var d := Data.near(pos, p) - pos
		if d.length_squared() < 1.0:
			continue
		b["pos"] = pos + d
		if is_instance_valid(b["mark"]):
			(b["mark"] as Node3D).global_position += d
		var n: Variant = b["node"]
		if is_instance_valid(n):
			n.global_position += d
			n.home += d
	for e: Dictionary in _extracts:
		var pos: Vector3 = e["pos"]

			(e["node"] as Node3D).global_position += d

func _update_bosses(p: Vector3) -> void:
	_boss_near = null
	if geo.fixed:
		_xb_update(p)
		return
	for b: Dictionary in _bosses:
		var pos: Vector3 = b["pos"]

		## 2026-11-20 数值框架：巢穴打空了十分钟（个人局内时间）：来新的镇守——重建标记、回到「没刷」，走近了照常刷

func _players_in(area: Rect2) -> int:

	if net_on():

			if is_instance_valid(r) and area.has_point(Vector2((r as Node3D).global_position.x, (r as Node3D).global_position.z)):
				n += 1
	return maxi(n, 1)

## 生成一只（双卫的第 sub 只）首领，我当主人
func _xb_spawn(b: Dictionary, sub: int) -> Node3D:
	var id := str(b["key2"]) if sub == 1 else str(b["id"])
	var spawn: Vector3 = b["pos2"] if sub == 1 else b["pos"]
	var arena: Rect2 = b["arena"]
	var n := _players_in(arena)
	var ri := int(b["region"])
	var lvl := Prog.boss_level(Prog.lair_tag(ri), _level_in(ri, spawn) + 2)
	# 老公式算的是一个人时的血量；按战斗区人数 × players_k(n) / players_k(1)，交给 ExBoss 的 players 换成相对量（之后人数变了它自己按比例改）
	var m := _spawn_mob(Data.ex_def(id), ri, spawn, false, lvl, ExMobs.players_k(n) / ExMobs.players_k(1))
	m.set_meta("boss_id", id)
	m.set_meta("lair", int(b["lid"]) * 2 + sub)
	m.set_meta("born", run_t)
	var nid: int = m.net_id
	var atk_pts := maxf(1.0, float(ExMobs.def(id).get("atk_pts", 10.0)))
	var br: Node3D = ExBoss.attach(m, id, arena, {
		"home": spawn, "y_range": b["yr"], "net": net,
		"fx_out": func(kind: String, p: Dictionary) -> void: _send_boss_fx(nid, kind, p),
		"summon_fn": func(sid: String, pos: Vector3, elite: bool) -> Node3D:
			if _mobs.size() >= mob_cap() + 10:
				return null
			var a := _spawn_mob(Data.ex_def(sid), int(geo.region_of(pos)), pos, elite)
			a.set_meta("summoned", true)
			_mobs.append(a)
			a.alert()
			return a,
		"night_fn": Callable(level, "is_night"),
		"room_n_fn": _room_count,
		"pts_k": m.atk / atk_pts,
		"players": n})
	if br != null and br.has_signal("event"):
		br.connect("event", _xb_event.bind(m))
	b["node2" if sub == 1 else "node"] = m
	return m


func _xb_event(kind: String, info: Dictionary, m: Node3D) -> void:

	match kind:

		"transform":
			hud.toast("%s 形态骤变！" % str(info.get("name", m.get("display_name"))), Color(0.85, 0.5, 1.0))


func _send_boss_fx(nid: int, kind: String, p: Dictionary) -> void:

## 生成一只 / 一对首领（我当主人）
func _xb_go(b: Dictionary, why: String) -> void:
	_xb_spawn(b, 0)
	if str(b["key2"]) != "":
		_xb_spawn(b, 1)
		var b0: Variant = (b["node"] as Node).get_node_or_null("ExBoss")
		var b1: Variant = (b["node2"] as Node).get_node_or_null("ExBoss")
		if b0 != null and b1 != null:
			ExBoss.link_duo(b0, b1)
	b["state"] = 1
	b["dead_n"] = 0
	var m: Node3D = b["node"]
	_announce_boss(m, "%s · %s" % [str(b["title"]), why])
	print("[raid] 新图首领 %s（%s）现身：%s Lv.%d 区域档 %d 血 %.0f 攻 %.1f" % [str(b["name"]), str(b["id"]), why, int(m.get("lv")), int(b["region"]),
			float(m.get("hp_max")), float(m.get("atk"))])


## 走近要刷：单机直接刷；联机先问房主，批给我才刷（net_ex_boss_owner）——两个人同时走近也只有一只
func _xb_request(b: Dictionary, why: String) -> void:
	if not net_on() or not net.has_method("ex_boss_claim"):
		_xb_go(b, why)
		return
	var own := int(b["owner"])
	if own != 0 and own != _my_net_id() and run_t - float(b["owner_t"]) < XB_OWNER_HOLD:
		return      # 别人名下刚有这只的消息：等他的木偶
	if run_t - float(b["claim_t"]) < XB_CLAIM_T:
		return
	b["claim_t"] = run_t
	b["why"] = why
	net.call("ex_boss_claim", int(b["lid"]), float(b["respawn"]))


## 房主判了首领 idx（= lid）：owner = 批给谁刷（> 0）；owner = 0 = 还在冷却（dead_t = 倒下的房间时刻）
func net_ex_boss_owner(idx: int, owner: int, dead_t: float) -> void:
	if not geo.fixed:
		return
	for b: Dictionary in _bosses:
		if int(b["lid"]) != idx:
			continue
		if owner <= 0:
			if dead_t > -1.0e8:
				net_ex_boss_dead(0, idx, dead_t, 0)
			return
		b["owner"] = owner
		b["owner_t"] = run_t
		if owner != _my_net_id():
			return
		if _xb_mine(b):
			return      # 已经是我刷的那只（重复批下来的）
		if not running or _xb_alive(b) or run_t - float(b["claim_t"]) > 10.0:
			# 批给我了却不刷（已经结算 / 场上已经有一只）：放手，别让别人一直等我
			if net_on() and net.has_method("ex_boss_release"):
				net.call("ex_boss_release", idx)
			return
		_xb_go(b, str(b.get("why", "有人闯入")))
		return


## 别人名下的首领倒下了（他那边广播的）：记冷却、亮镇守阵
func net_ex_boss_dead(_from: int, idx: int, t: float, killer: int) -> void:
	if not geo.fixed:
		return
	for b: Dictionary in _bosses:
		if int(b["lid"]) != idx:
			continue
		var was := int(b["state"])
		b["state"] = 2
		b["back"] = t + float(b["respawn"])
		b["killer"] = killer
		if was != 2:
			clears += 1
		_light_boss_gate(idx)


## 别人名下的首领（木偶）死了（net_mob_died 的 lair ≥ 0）：先记上冷却，免得这一跳我这边又刷一只（主人随后还会发 ex_boss_dead 带准确的时刻）
func _xb_gone(lair: int) -> void:
	var b := _xb_by_lair(lair)
	if b.is_empty():
		return
	var k := "node2" if lair % 2 == 1 else "node"
	var cur: Variant = b.get(k)
	if cur == null or not is_instance_valid(cur) or bool(cur.get("puppet")):
		b[k] = null
	if not _xb_alive(b):
		b["state"] = 2
		b["back"] = maxf(float(b["back"]), world_t() + float(b["respawn"]))
		_light_boss_gate(int(b["lid"]))


## 我名下的首领（或双卫的一只）倒下：两只都倒了才掉老巢穴首领的遗藏（_loot + 残片 + 霸主精魄），记冷却、广播、亮镇守阵
func _xb_died(m: Node3D, pos: Vector3, dep: float, tag: int, zb: float) -> void:
	var lair := int(m.get_meta("lair", -1))
	var b := _xb_by_lair(lair)
	if b.is_empty():
		return
	b["node2" if lair % 2 == 1 else "node"] = null
	var killer := int(m.get_meta("hitter", _my_net_id()))
	b["dead_n"] = int(b["dead_n"]) + 1
	if str(b["key2"]) != "" and _xb_alive(b):
		hud.toast("%s 倒下，另一只还在！" % str(m.get("display_name")), Color(1.0, 0.7, 0.4))
		return
	var ri := int(b["region"])
	var duo := str(b["key2"]) != ""
	var loot := _loot(ri, 6 if duo else 5, (1.4 if duo else 1.3) + zb, dep, tag)
	loot.append(Items.frag(Prog.frag_to(ri)))      # 巢穴首领的遗藏里另有一片本档的破境丹残片
	var lt := Prog.lair_tag(ri)
	if _rng.randf() < Prog.OVERLORD_P[clampi(lt, 0, Prog.OVERLORD_P.size() - 1)]:
		loot.append(Items.overlord(lt))
	_drop_chest(pos, loot, "%s 的遗藏" % str(b["name"]))
	var now := world_t()
	b["state"] = 2
	b["back"] = now + float(b["respawn"])
	b["killer"] = killer
	clears += 1
	_light_boss_gate(int(b["lid"]))
	if net_on() and net.has_method("ex_boss_dead"):
		net.call("ex_boss_dead", int(b["lid"]), now, killer)
	hud.banner("击败 %s" % str(b["name"]), "遗藏已现，速取　约 %d 分钟后会有新的镇守" % int(ceilf(float(b["respawn"]) / 60.0)), Color(1.0, 0.85, 0.4))
	print("[raid] 新图首领 %s 倒下：遗藏 %d 件，%d 秒后重生" % [str(b["name"]), loot.size(), int(b["respawn"])])


## 0.4 秒一次：冷却到了回到未刷；走近了要刷；主人走了 / 木偶没了回到未刷；近处的交给界面
func _xb_update(p: Vector3) -> void:
	var now := world_t()
	for b: Dictionary in _bosses:
		var st := int(b["state"])
		if st == 2 and now >= float(b["back"]):
			b["state"] = 0
			st = 0
		var pos: Vector3 = b["pos"]
		var d := pos.distance_to(p)
		if st == 0 and d < XB_SPAWN_R and not _xb_alive(b) and running:
			_xb_request(b, "有人闯入")
		elif st == 1 and not _xb_alive(b):
			b["state"] = 0 if now >= float(b["back"]) else 2
		if _xb_alive(b) and d < XB_NEAR_R:
			var n0: Variant = b["node"]
			_boss_near = n0 if n0 != null and is_instance_valid(n0) and bool(n0.is_alive()) else b["node2"]

##   晚上用户又说「怪还是太稀疏，打得不爽」，实测（raid_pace_test.gd）40 米内平均只有 1.4 ~ 2.5 只，杀一只就得再飞三五十米找下一只
##   → 15 / 22，一拨小怪 3 ~ 5 只、近处那圈挪到 28 ~ 50 米，导演点数涨得快一倍。仍然只有两成一出来就扑。
##   联机时身边 160 米内有 n 个真人（各自有导演、名下的怪大家都看得见打得着）：每人只刷 1/(n+1)，凑在一起的一群人身边总数还是一个人的量
##   （三开自检实测：按 1/(1+0.6n) 分时三个人凑一块，每人眼前四五十只，太挤）
const MOB_CAP := 22
## 2026-11-20 夜：用户要手机端的怪再多一点 → 15 → 20（实测每只怪手机档渲染 + 逻辑约 0.05 ms，见 raid_pace_test.gd 的 --pace-perf）
const MOB_CAP_MOBILE := 20


var cap_override := -1   # 只给节奏探针用（raid_pace_test.gd 的 --pace-cap=N：同一份代码量改前 / 改后的上限）


func mob_cap() -> int:
	var base := cap_override if cap_override > 0 else (MOB_CAP_MOBILE if mobile else MOB_CAP)
	if _near_mates <= 0:
		return base
	return maxi(roundi(float(base) / (1.0 + float(_near_mates))), 5)


var _near_mates := 0      # 身边 160 米内的其他真人（_director 里 0.4 秒数一次）


func _count_near_mates(p: Vector3) -> int:
	if not net_on():
		return 0
	var n := 0
	for id: int in net.remotes.keys():
		var r: Variant = net.remotes[id]
		if is_instance_valid(r) and (r as Node3D).global_position.distance_to(p) < 160.0:
			n += 1
	return n


## 联机时身边来了人、我的上限变小了：每 0.4 秒收走一只 90 米外还没发现人的普通怪（精英、首领不动），一群人凑一块以后十来秒就降到一个人的量
func _trim_far(p: Vector3) -> void:
	var far: Variant = null
	var fd := 90.0
	for m: Variant in _mobs:
		if not is_instance_valid(m) or m.boss or m.elite or m.state != Mob.S.IDLE or m.has_meta("camp"):
			continue
		var d: float = m.global_position.distance_to(p)
		if d > fd:
			fd = d
			far = m
	if far != null:
		_mobs.erase(far)
		far.queue_free()


func _director(p: Vector3) -> void:
	var hold := _credits < -1000.0 or _wave_t > 900.0      # 自检压着导演（点数压成负的 / 波次计时拉满）：下面会改写 _wave_t，先读
	var cap := mob_cap()
	var i := _mobs.size() - 1
	while i >= 0:
		var m: Variant = _mobs[i]
		if not is_instance_valid(m):
			_mobs.remove_at(i)
		elif not m.has_meta("camp") and m.global_position.distance_to(p) > (260.0 if m.boss else 150.0) and (m.state == Mob.S.IDLE or m.state == Mob.S.CHASE):
			m.queue_free()
			_mobs.remove_at(i)
		i -= 1
	if net_on() and not net.remotes.is_empty():
		_retarget_mobs(p)
	_near_mates = _count_near_mates(p)
	cap = mob_cap()
	if _mobs.size() > cap + 2:
		_trim_far(p)
	# 2026-11-20 数值框架：攒点数的快慢看主角所在处的深度（原来随局内时间涨）；晚上加密度：2.2 → 4.4，存顶 280 → 420
	_credits = minf(_credits + 4.4 * (1.0 + 0.4 * (1.0 + here_depth)) * 0.4, 420.0)
	_wave_t -= 0.4
	_elite_t -= 0.4
	if _mobs.size() <= cap / 3:
		_wave_t = minf(_wave_t, 1.0)      # 周围快空了（清完了 / 飞远了）：马上补
	_event_t -= 0.4
	if _event_t <= 0.0 and _boss_near == null and not geo.fixed and Data.land_r(p.x, p.z) < Data.PLAY_R:      # 外海上不来游荡首领（新图没有「天地异动」：首领都在巢穴里）
		_event_t = _rng.randf_range(240.0, 360.0)
		boss_event()
	if camps != null and camps.on:
		# 固定刷怪点（raid_camps.gd）：走近了把点上活着的怪建出来，急飞时不建；自检把点数压成负的 / 波次计时拉满 = 不要刷怪
		camps.tick(p, hold)
		return
	if _wave_t > 0.0 or _mobs.size() >= cap or Data.land_r(p.x, p.z) > Data.PLAY_R:      # 外海上不刷怪
		return
	# 没到上限的六成时两三秒一拨、尽快把周围铺起来；之后慢慢补（点数够不够另算）
	_wave_t = _rng.randf_range(2.0, 3.5) if _mobs.size() < cap * 6 / 10 else _rng.randf_range(4.0, 7.0)
	spawn_wave(p)


## 搜点的守卫：在 pos 周围 8~14 米刷 n 只该区域的怪（不花导演点数，可以超过同屏上限一点）；alert = 一出来就扑向主角
func spawn_guards(pos: Vector3, ri: int, n: int, elite: bool, alert: bool) -> void:
	if _mobs.size() >= mob_cap() + 3:
		return
	if geo.fixed:
		_spawn_guards_fixed(pos, ri, n, elite, alert)
		return
	var pool: Array = Data.REGIONS[ri]["mobs"]
	for i in n:
		var a := _rng.randf() * TAU
		var q := pos + Vector3(cos(a), 0.0, sin(a)) * _rng.randf_range(8.0, 14.0)
		q.y = _ground(q.x, q.z)
		var m := _spawn_mob(int(pool[_rng.randi() % pool.size()]), ri, q, elite and i == 0)
		m.home = pos
		_mobs.append(m)
		Vfx.takeoff_burst(q, (m.def["cols"] as Array)[2], 1.2 if elite and i == 0 else 0.7)
		if alert:
			m.alert()
	if elite:
		hud.toast("精英守卫现身", Color(1.0, 0.7, 0.3))


## 新图的搜点守卫：本图的怪（ex_mobs 的刷怪池），落点问关卡（同一层、不进墙）
func _spawn_guards_fixed(pos: Vector3, ri: int, n: int, elite: bool, alert: bool) -> void:
	var pool: Array = geo.mob_pool(pos)
	var made := 0
	for i in n:
		var q: Vector3 = geo.spot_near(pos, 6.0, 12.0, _rng)
		if not q.is_finite():
			continue
		var m := _spawn_mob(int(pool[_rng.randi() % pool.size()]), ri, q, elite and made == 0)
		m.home = pos
		_mobs.append(m)
		made += 1
		Vfx.takeoff_burst(q, (m.def["cols"] as Array)[2], 1.2 if elite and made == 1 else 0.7)
		if alert:
			m.alert()
	if elite and made > 0:
		hud.toast("精英守卫现身", Color(1.0, 0.7, 0.3))


## 花点数在 p 周围买一拨怪；返回买了几只
func spawn_wave(p: Vector3) -> int:
	# 一拨 = 一小群，落在随机方向上的一个点周围。六成在近处（28 ~ 50 米，都在它们发现人的 26 米之外），
	# 四成在远处（50 ~ 90 米）：放眼望去远近都有，不是只有身边那一圈
	var a := _rng.randf() * TAU
	var dist := _rng.randf_range(28.0, 50.0) if _rng.randf() < 0.6 else _rng.randf_range(50.0, 90.0)
	var centre := p + Vector3(cos(a), 0.0, sin(a)) * dist
	# 2026-11-20 数值框架：刷哪个区域的怪、几级，按刷新点算（不按主角站的地方：边界上不会刷出隔壁区域的怪）
	var src := centre if Data.land_r(centre.x, centre.z) < Data.PLAY_R else p      # 刷新点落到海边外面了：还按主角那里算
	var ri := Data.region_at(src.x, src.z)
	var pool: Array = Data.REGIONS[ri]["mobs"]
	var idx: int = pool[_rng.randi() % pool.size()]
	var cost: float = Data.SHAPE_BASE[str(Data.MOBS[idx]["shape"])][2]
	# 精英：2026-11-20 晚改成按时间来（40 ~ 70 秒一只，带两只随从）。原来是每拨 22% 的几率、要攒够 6 倍点数，
	# 怪加密以后点数总被一拨拨小怪花光，实测精英只剩击杀的 1%
	var elite := _elite_t <= 0.0 and _credits >= cost * 3.0
	if elite:
		_elite_t = _rng.randf_range(40.0, 70.0)
		cost *= 3.0
	var n := 0
	# 一拨几只（2026-11-20 晚加密度）：小怪 3 ~ 5 只一群，巨汉 1 ~ 2 只；精英单独一只，身边再跟两只同区域的小怪
	var big := str(Data.MOBS[idx]["shape"]) == "brute"
	var want := 1 if elite else (_rng.randi_range(1, 2) if big else _rng.randi_range(3, 5))
	while _credits >= cost and _mobs.size() < mob_cap() and n < want:
		var aa := _rng.randf() * TAU
		var pos := centre + Vector3(cos(aa), 0.0, sin(aa)) * _rng.randf_range(2.0, 4.0 + 1.2 * float(want))
		pos.y = _ground(pos.x, pos.z)
		var m := _spawn_mob(idx, ri, pos, elite)
		_mobs.append(m)
		_credits -= cost
		n += 1
		Vfx.takeoff_burst(pos, (m.def["cols"] as Array)[2], 0.7 if not elite else 1.3)
		if _rng.randf() < (0.5 if elite else 0.2):
			m.alert()      # 少数一出来就扑过来，其余的在原地转悠，等人走近
	if elite and n > 0:
		# 精英的随从：两只小怪（不扣点数、不超上限太多）
		var small: Array = (Data.REGIONS[ri]["mobs"] as Array).filter(func(j: int) -> bool: return str(Data.MOBS[j]["shape"]) != "brute")
		if not small.is_empty():
			for i in 2:
				if _mobs.size() >= mob_cap() + 2:
					break
				var aa := _rng.randf() * TAU
				var pos := centre + Vector3(cos(aa), 0.0, sin(aa)) * _rng.randf_range(3.0, 6.0)
				pos.y = _ground(pos.x, pos.z)
				var f := _spawn_mob(int(small[_rng.randi() % small.size()]), ri, pos)
				_mobs.append(f)
				n += 1
		hud.toast("精英 %s 出现了" % Data.MOBS[idx]["name"], Color(1.0, 0.7, 0.3))
	return n


func _on_mob_died(m: Node3D) -> void:
	var cd := int(m.get_meta("camp", -1))
	if cd >= 0 and camps != null:
		camps.on_dead(cd / 8, cd % 8, -1.0, m.slain_by_rival)
	if net_on():
		net.send_mob_dead(m.net_id, m.global_position, m.lv, m.xp_k, int(m.get_meta("lair", -1)) if cd < 0 else -2 - cd, int(m.get_meta("hitter", 0)))
	var ri: int = m.region
	var pos: Vector3 = m.global_position
	pos.y = _floor(pos.x, pos.z, pos.y + 0.5)      # 大世界 = 地形；新图按脚高找落脚处（楼上死的别掉到楼下）
	var by_rival: bool = m.slain_by_rival
	var zk: Dictionary = geo.zone_k(pos)
	var zb := float(zk["bonus"]) if geo.fixed else 0.0      # 新图的掉落加成（第 3 节）
	# 联机：刷怪点的怪由离得近的人建，主人不一定在打它——离我 100 米外死的不算我的（打它的人照常按 share 分）
	if not by_rival and (cd < 0 or not net_on() or geo.near(pos, player.global_position).distance_to(player.global_position) < 100.0):
		kills += 1
		# 2026-11-20 数值框架：攒心得，撤离成功才化为修为；光点从怪身上飞过来。
		# 2026-11-23：心得按怪的区域档；带人闸——打过它的别人里境界最高的高出此地、也高出我，心得打折（net_mob_take 记的 hi_realm）
		var ck := _carry_k(int(m.get_meta("hi_realm", -1)), ri)
		if geo.fixed:
			ck *= float(zk["xp"])      # 新图的心得倍率
		add_insight(m.lv, m.xp_k * ck, m.center(), 2 if m.boss else (1 if m.elite else 0), ri)
	if geo.fixed and bool(m.get_meta("summoned", false)):
		return      # 首领召唤的小怪：不掉东西
	var dep := depth_of(ri, pos)       # 掉落按怪倒下的地方的深度
	var tag := Prog.mob_tag(ri, m.lv)  # 2026-11-23：掉的修为丹是这只怪的境界档（不按倒下处现算：首领、跨档线的怪掉自己那一档的）
	if geo.fixed and m.boss and m.has_meta("boss_id"):
		_xb_died(m, pos, dep, tag, zb)      # 新图巢穴首领（双卫两只都倒了才掉；归属 / 冷却走房主仲裁）
		return
	if m.boss:
		var lair_ri := -1
		var lair_id := -1
		for b: Dictionary in _bosses:
			if b["node"] == m:
				lair_ri = int(b["region"])
				lair_id = int(b["lid"])
		if lair_ri >= 0:
			_lair_cleared(lair_id)
		var loot := _loot(ri, 5 if lair_ri >= 0 else 3, (1.3 if lair_ri >= 0 else 0.8) + zb, dep, tag)
		if lair_ri >= 0:
			loot.append(Items.frag(Prog.frag_to(lair_ri)))      # 巢穴首领的遗藏里另有一片本区的破境丹残片
			## 2026-11-23 照 Hammer 的霸主精魄：按巢穴首领的档（lair_tag）的几率多掉一颗，境界 ≥ 该档就能服
			var lt := Prog.lair_tag(lair_ri)
			if _rng.randf() < Prog.OVERLORD_P[clampi(lt, 0, Prog.OVERLORD_P.size() - 1)]:
				loot.append(Items.overlord(lt))
		_drop_chest(pos, loot, "%s 的遗藏" % m.display_name)
		hud.banner("击败 %s" % m.display_name, "遗藏已现，速取" + ("　巢穴已空，约十分钟后会有新的镇守" if lair_ri >= 0 else ""), Color(1.0, 0.85, 0.4))
	elif m.elite:
		_drop_chest(pos, _loot(ri, 2, 0.5 + zb, dep, tag), "精英的遗物")
	elif _rng.randf() < (0.2 if by_rival else KILL_DROP):
		for it: Dictionary in _loot(ri, 1, 0.0 + zb, dep, tag):
			_drop_item(pos + Vector3(_rng.randf_range(-0.8, 0.8), 0.0, _rng.randf_range(-0.8, 0.8)), it, 0.5)


## 掉落：2026-11-20 数值框架——品阶运气、修为丹品质按出产处的深度（交给 Data.roll_loot），价值 × Prog.value_mult(深度)（不再随局内时间涨）
## 2026-11-23：pill_realm = 掉出来的修为丹是哪个境界的（怪的境界档）；-1 = 按掉落处（Prog.pill_realm_at）
func _loot(ri: int, n: int, bonus: float, depth := 0.0, pill_realm := -1) -> Array:
	var items := Data.roll_loot(ri, n, _rng, bonus, depth, pill_realm)
	var k := Prog.value_mult(depth)
	for it: Dictionary in items:
		it["value"] = int(roundf(float(it["value"]) * k))
	return items


## 搜点搜出来的东西（raid_search.gd，品阶 / 品质已按该点的深度掷过）：价值 × 该点深度的 Prog.value_mult，
## 在搜点旁边（朝主角那一侧）放一只「搜得」的箱子，和怪掉的箱子一样走拾取界面（2026-11-20 起不自动进背包）
func site_loot(items: Array, pos: Vector3, from: String) -> void:
	var k := Prog.value_mult(depth_of(geo.region_of(pos) if geo.fixed else Data.region_at(pos.x, pos.z), pos))
	var best := 0
	for it: Dictionary in items:
		it["value"] = int(roundf(float(it["value"]) * k))
		best = maxi(best, int(it["tier"]))
	var to_p := Vector3(player.global_position.x - pos.x, 0.0, player.global_position.z - pos.z)
	var at := pos + (to_p.normalized() * minf(1.8, to_p.length() * 0.6) if to_p.length() > 0.3 else Vector3(1.4, 0.0, 0.0))
	at.y = _floor(at.x, at.z, pos.y + 0.5)
	_drop_chest(at, items, "%s · 搜得" % from)
	hud.toast("%s：搜出 %d 件，点「拾取」收进背包" % [from, items.size()], Data.TIER_COLORS[best])
	Sfx.play("perfect", -10.0, 1.2 + 0.1 * float(best))

func _engaged() -> int:
	var n := 0
	for f: Variant in _rivals:
		if is_instance_valid(f) and f.state != AirFoe.S.IDLE and f.state != AirFoe.S.DEAD:
			n += 1
	return n


func _engage_ok(_f: Node3D) -> bool:
	return run_t > 40.0 and _engaged() < 2


## 给修士找一只附近的怪（没被别的修士盯上的小怪优先）
func _prey_for(f: Node3D) -> Node3D:
	var best: Node3D = null
	var bd := 48.0
	for m: Variant in _mobs:
		if not is_instance_valid(m) or not m.is_alive() or m.boss:
			continue
		if m.decoy != null and is_instance_valid(m.decoy) and m.decoy != f:
			continue
		var d: float = m.global_position.distance_to(f.global_position)
		if d < bd:
			bd = d
			best = m
	return best


func spawn_rival(pos: Vector3, hostile: bool) -> Node3D:
	var ri: int = geo.region_of(pos) if geo.fixed else Data.region_at(pos.x, pos.z)
	var f := AirFoe.new()
	f.player = player
	f.display_name = _names[0]
	_names.append(_names.pop_front())
	f.raid = true
	f.pro = true
	f.roam = true
	f.hostile = hostile
	f.lv = _level_in(ri, pos)      # 2026-11-20：和那里的怪同级
	## 2026-11-23 数值重构：当成同区同级的「怪」——外观 / 对波档位按区域境界档（不再跟着主角的境界走），
	## 气血 = RIVAL_HP × HP_CONV × 血量级，打怪 = AI_PREY_K × 同级主角一单位 × 攻量级（打主角的倍数在 _update_rivals 按量级比现算）
	f.look_i = Prog.mob_tag(ri, f.lv)
	f.style = 2 if _rng.randf() < 0.5 else 0

	f.hp_max = Data.RIVAL_HP * CS.HP_CONV * CS.mob_s_hp(ri, f.lv)
	f.prey_dmg = AI_PREY_K * CS.unit_same(1) * CS.mob_s_atk(ri, f.lv)
	f.volley_init = false      # 不主动朝主角连发飞剑（也就不会动不动拉人对波）；主角先朝它连发，它才回敬
	f.roam_skill = Data.RIVAL_SKILL
	f.roam_near = 0.2          # 各逛各的：偶尔才往主角这边来，而且隔着四五十米以上（原来五成半的时候都往主角身边 18 ~ 50 米凑）
	f.roam_far = 2.2
	f.dmg_in = player_dmg_vs.bind(f)      # 2026-11-23：防御按它自己的区域档 / 等级（meta region）
	f.prey_fn = _prey_for
	f.engage_ok = _engage_ok
	f.floor_fn = _ground
	f.position = pos
	f.set_meta("region", ri)
	f.died.connect(_on_rival_died)
	_mob_root.add_child(f)
	_rivals.append(f)
	return f


func _update_rivals(p: Vector3) -> void:
	var i := _rivals.size() - 1
	var idle_far: Node3D = null
	while i >= 0:
		var f: Variant = _rivals[i]
		if not is_instance_valid(f) or f.state == AirFoe.S.DEAD:
			_rivals.remove_at(i)
		else:

func _ally_prey(f: Node3D) -> Node3D:

## 2026-11-23：现在能服的修为丹（本境丹 / 够境界的霸主精魄）比价时按 ×3 算——别为了腾地方放过一颗能吃的丹
func _swap_out(it: Dictionary) -> int:
	var need := Items.size_of(it)
	var dens := float(it["value"]) * (3.0 if _edible(it) else 1.0) / float(need)
	var best := -1
	var bd := dens / 1.25
	var free := Data.BAG_SIZE - bag_used()
	for i in bag.size():
		var b: Dictionary = bag[i]
		if not ["gear", "mat", "herb"].has(str(b["kind"])):
			continue
		var bs := Items.size_of(b)
		if free + bs < need:
			continue
		var d := float(b["value"]) / float(bs)
		if d < bd:
			bd = d
			best = i
	return best

func _take(it: Dictionary, quiet := false) -> bool:
	if not _fits(it):

		hud.toast("背包满了：丢下 %s（%d），换 %s（%d）" % [old["name"], int(old["value"]), it["name"], int(it["value"])], Color(0.95, 0.8, 0.5))
	bag.append(it)
	if not quiet:
		var col: Color = Data.TIER_COLORS[int(it["tier"])]
		var tw := Items.tier_word(it)      # 修为丹名字里带了品质，不再跟一个「凡品」
		hud.toast(("拾取 %s · %s（%s）" % [it["name"], tw, Items.badge(it)]) if tw != "" else ("拾取 %s（%s）" % [it["name"], Items.badge(it)]), col)
		Sfx.play("perfect", -12.0, 1.3 + 0.1 * float(it["tier"]))
	if _need_key >= 0 and str(it["kind"]) == "key" and _trib_to < 0 and not _in_trib:
		_bottleneck(_need_key)
	return true

## 界面上列出来（名字、品阶、箱子里有什么）+「拾取」按钮，玩家自己点（键盘 T 拾取最近的一件）。

## 拾取卡片的列表（raid_hud.gd 一件一张）：[掉落物, 箱子里第几件（单件是 -1）, 物品]，近的在前
func loot_entries() -> Array:
	var out: Array = []
	for pk: Variant in near_loot:
		if not is_instance_valid(pk):
			continue
		if pk.is_chest():
			var items: Array = pk.items
			for j in items.size():
				out.append([pk, j, items[j]])
		else:
			out.append([pk, -1, pk.item])
	return out


## 拾取：pk 是地上的一件 / 一个箱子；idx = 箱子里的第几件（-1 = 全部）。联机共享的掉落走房主判（_claim，整个拿）
func pick_up(pk: Node3D, idx := -1) -> void:
	if not running or not is_instance_valid(pk) or not _pickups.has(pk):
		return
	if _shared(pk):
		_claim(pk)
		return
	if not pk.is_chest():
		if _take(pk.item):
			_remove_pickup(pk)
		else:
			hud.toast("背包已满（打开背包丢弃、服用，或放进储物戒）", Color(1.0, 0.5, 0.4))
		return
	var items: Array = pk.items
	var order: Array = [idx] if idx >= 0 else range(items.size())
	var taken: Array = []
	for j: int in order:
		if j >= items.size():
			continue
		if _take(items[j]):
			taken.append(j)
		elif idx >= 0 or _full_t <= 0.0:
			_full_t = 2.0
			hud.toast("背包已满，没拿完的还在箱子里", Color(1.0, 0.5, 0.4))
			break
	taken.sort()
	taken.reverse()
	for j: int in taken:
		items.remove_at(j)
	if items.is_empty():
		_remove_pickup(pk)


func _remove_pickup(pk: Node3D) -> void:
	_pickups.erase(pk)
	near_loot.erase(pk)
	Vfx.takeoff_burst(pk.global_position, Data.TIER_COLORS[clampi(int(pk.tier()), 0, 4)], 0.4)
	pk.queue_free()


## 键盘 T：拾取最上面那张卡片（离得最近的那一件）
func pick_nearest() -> void:
	var es := loot_entries()
	if not es.is_empty():
		pick_up(es[0][0], int(es[0][1]))

func _build_extract(pos: Vector3, active: bool, beam_col := Color(0.3, 0.95, 1.0)) -> Node3D:
	var root := Node3D.new()
	add_child(root)
	root.global_position = pos
	var stone := MeshLib.std_mat(Color(0.6, 0.62, 0.66), 0.9)
	var base := MeshInstance3D.new()
	var cm := CylinderMesh.new()
	cm.top_radius = EXTRACT_R
	cm.bottom_radius = EXTRACT_R + 0.6
	cm.height = 0.5
	cm.radial_segments = 16
	base.mesh = cm
	base.material_override = stone
	base.position.y = 0.1
	root.add_child(base)
	for i in 6:
		var a := TAU * float(i) / 6.0
		var broken := not active and i % 2 == 0
		var pm := MeshInstance3D.new()
		var bm := BoxMesh.new()
		bm.size = Vector3(0.9, 1.6 if broken else 5.0, 0.9)
		pm.mesh = bm
		pm.material_override = stone
		pm.position = Vector3(cos(a), 0.0, sin(a)) * (EXTRACT_R - 0.6) + Vector3(0.0, bm.size.y * 0.5, 0.0)
		pm.rotation = Vector3(0.0, -a, 0.25 if broken else 0.0)
		root.add_child(pm)
	if active:
		_beam(root, beam_col, 220.0, 3.0)
		var glow := MeshInstance3D.new()
		var gm := CylinderMesh.new()
		gm.top_radius = EXTRACT_R - 1.2
		gm.bottom_radius = EXTRACT_R - 1.2
		gm.height = 0.08
		gm.radial_segments = 16
		glow.mesh = gm
		var m := StandardMaterial3D.new()
		m.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED
		m.albedo_color = beam_col.darkened(0.35)
		glow.material_override = m
		glow.position.y = 0.4
		root.add_child(glow)
	return root


## 最近的撤离点：{pos, active, known, name, dist}

		hud.banner("与房主断开", "没能撤离：本局所得留不住，带进来的法宝和随身物已还回洞府", Color(1.0, 0.5, 0.4))
	print("[raid] 断线（没收到关房通知）：按中途退出处理，背包 %d 件留不住，心得 %.0f 留下 %.0f" % [bag.size(), ins, keep])
	await get_tree().create_timer(4.5, true, false, true).timeout
	player.melee.hp = player.melee.HP_MAX
	level.exit_requested.emit()

func _abort_trib() -> void:
	if not _in_trib:
		return
	if is_instance_valid(_trib):
		if _trib.finished.is_connected(_on_trib_done):
			_trib.finished.disconnect(_on_trib_done)
		_trib._end()
	_trib_to = -1
	_trib_free = false
	_end_trib()

func _finish(success: bool) -> void:
	if not running:
		return
	running = false
	_mob_root.process_mode = Node.PROCESS_MODE_DISABLED
	bag.append_array(_eat_q)      # 一键服丹还没服完的放回背包
	_eat_q.clear()
	var value := bag_value()
	var items := bag.duplicate()
	var lost_gear := gear.duplicate()
	if not success and net_on() and not (items + gear).is_empty():
		# 联机里陨落：背包和法宝变成一只储物袋留在原地，房间里谁先开归谁
		var dp := player.global_position
		_drop_chest(Vector3(dp.x, _floor(dp.x, dp.z, dp.y + 0.5), dp.z), items + gear, "%s 的储物袋" % net.my_name)
	## 2026-11-20 数值框架：心得结算——撤离成功全数化为修为（2026-11-23 起到了瓶颈照收、突破时结转，只有结转也存满了的才散掉）；陨落 / 困死只留 Prog.DEATH_KEEP 成（现在是 0）。
	## 房主关房（room_collapsed）照撤离成功走这里；没收到关房通知就断线的（room_lost）不走这里，心得 × DEATH_KEEP。修为先写进存档，end_raid 不再传 lv / xp
	if _ins_fx != null and is_instance_valid(_ins_fx):
		_ins_fx.clear()
	var ins := insight
	var keep := ins if success else ins * Prog.DEATH_KEEP
	var ins_gain := 0.0
	var ins_ups := 0
	if keep > 0.0:
		var st := Prog.profile_add_xp(keep)
		ins_gain = float(st["used"])
		ins_ups = int(st["ups"])
	insight = 0.0
	insight_shown = 0.0
	_sync_cult()
	var res := Profile.end_raid(success, items, ring, gear, player.realm_idx, -1, 0.0)
	stash_total = Profile.lingshi()
	if success:
		Vfx.breakthrough(player.global_position, Color(0.4, 1.0, 1.0), 1.0)
		Sfx.play("ascend", -2.0, 1.0)
	else:
		bag.clear()
		gear.clear()
		Combat.slowmo(0.15, 0.6, 0.8)
		Vfx.death_burst(player.center(), Color(1.0, 0.3, 0.25))
	hud.show_result(success, items, value, kills, stash_total, {"res": res, "lost_gear": lost_gear, "ring": ring.duplicate(),
			"ins": ins, "ins_gain": ins_gain, "ins_ups": ins_ups, "ins_lost": ins - ins_gain, "lv": lv})
	print("[raid] 结算：%s 带出 %d 件 价值 %d 击杀 %d 洞府灵石 %d 丢失法宝 %d 储物戒 %d 件 心得 %.0f → 修为 +%.0f（升 %d 级，散去 %.0f）Lv.%d" % [
			"撤离成功" if success else "陨落", items.size() if success else 0,
			value if success else 0, kills, stash_total, 0 if success else lost_gear.size(), ring.size(), ins, ins_gain, ins_ups, ins - ins_gain, lv])
	await get_tree().create_timer(4.5, true, false, true).timeout
	player.melee.hp = player.melee.HP_MAX
	level.exit_requested.emit()


func _process(dt: float) -> void:
	if not running:
		return
	_sync_shown()
	_watch_cine()
	if _in_trib:
		# 雷劫期间：局内时间、刷怪、拾取、撤离都停住，只管天色（新图的天色归 ex_level 管）
		if not geo.own_atmo:
			_blend_atmosphere(dt)
		# 元婴雷劫后面还接着一段结婴过场：过场放完再解冻怪物、恢复界面
		if not is_instance_valid(_trib) and YuanyingCine.playing == null:
			_end_trib()
		return
	## 2026-11-23：渡劫失败后 _trib_hold 秒内不起劫；身上还有本境丹时也缓几秒、一键服丹排着队时等服完（突破以后就服不了了）。GM 渡劫不等
	if _trib_hold > 0.0:
		_trib_hold -= dt
		if _trib_hold <= 0.0 and _trib_to < 0 and _at_cap():
			# 2026-11-23：渡劫失败后修为结转还在、照样卡在瓶颈——不自己再起劫（不然隔一分钟就扣一次布阵灵石），等玩家服丹时由 _cult_changed 再起
			hud.toast("心境已稳，服一颗丹即可再引雷劫", Color(1.0, 0.86, 0.45))
	if _trib_to >= 0 and (_trib_free or (_trib_hold <= 0.0 and _eat_q.is_empty())):
		_try_start_trib()
		if _in_trib:
			return
	run_t += dt
	_solo_t += dt
	_full_t -= dt
	if geo.own_atmo:
		_fixed_region_tick(player.global_position)
	else:
		_blend_atmosphere(dt)
	_update_pickups(dt)
	if camps != null:
		camps.frame()
	# 一键服丹：一颗颗出队。2026-11-23：瓶颈时照服（结转），结转也存满了才停、剩下的放回背包；排队时境界变了服不了的也放回去
	if not _eat_q.is_empty():
		_eat_t -= dt
		if _eat_t <= 0.0:
			_eat_t = EAT_GAP
			if _xp_full():
				hud.toast("修为已满，先去渡劫，余下 %d 颗放回背包" % _eat_q.size(), Color(1.0, 0.75, 0.45))
				bag.append_array(_eat_q)
				_eat_q.clear()
			else:
				var it: Dictionary = _eat_q.pop_front()
				if _edible(it):
					_eat_one(it)
				else:
					bag.append(it)
	var hurt_now := player.melee.hp < _last_hp - 0.01
	_update_extract(dt)
	if not running:
		return
	# 搜点读条（法宝「识」加快）
	search_k = search.step(dt, player.global_position, hurt_now, 1.0 + float(fx["seek"])) if is_instance_valid(search) else 0.0
	if player.melee.hp <= 0.0:
		_finish(false)
		return
	# 脱战 5 秒后缓慢回血（法宝「养」加快）
	if player.melee.hp < _calm_hp - 0.01:
		_calm_t = 5.0
	_calm_hp = player.melee.hp
	_calm_t -= dt
	# 2026-11-20 夜：脱战回复 1.5 → CS.CALM_REGEN（4）每秒；空战时由空战自己的回复（6 / 秒）管，这里不再叠上去
	if _calm_t <= 0.0 and player.melee.hp < player.melee.HP_MAX:
		# 空战里基础那份由空战自己回（6 / 秒），这里只补「养」法宝的加成
		var base_regen := 0.0 if player.st == Player.St.AIR else CS.CALM_REGEN
		player.melee.hp = minf(player.melee.hp + (base_regen + CS.CALM_REGEN * float(fx["vital"]) * 1.5) * dt, player.melee.HP_MAX)
		_calm_hp = player.melee.hp
		_last_hp = maxf(_last_hp, player.melee.hp)
	reveal_t = maxf(reveal_t - dt, 0.0)
	for i in range(pings.size() - 1, -1, -1):
		pings[i][2] = float(pings[i][2]) - dt
		if float(pings[i][2]) <= 0.0:
			pings.remove_at(i)
	_tick -= dt
	if _tick <= 0.0:
		_tick = 0.4
		var p := player.global_position
		if geo.loop:
			_reimage(p)
		_update_here(p)
		if is_instance_valid(search):
			search.reveal_t = reveal_t
			search.tick(p, float(fx["seek"]))
		_update_bosses(p)
		_director(p)
		_update_rivals(p)
		_update_allies(p)
		_timeline()
		if not running:
			return
	hud.boss = _boss_near if is_instance_valid(_boss_near) else null
