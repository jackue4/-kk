## 从任何地方打开（hud / level_select 之外的入口用）：挂到 host 下、翻到 tab、暂停树放鼠标，关掉时恢复
static func open(host: Node, tab: Variant = "prep") -> Control:
	var scr: GDScript = load(SELF_PATH)
	var ui: Control = scr.new()
	ui.set("font", UiStyle.kai())
	host.add_child(ui)
	ui.call("_set_tab", tab)
	var tree := host.get_tree()
	var was := tree.paused
	tree.paused = true
	var mm := Input.mouse_mode
	Input.mouse_mode = Input.MOUSE_MODE_VISIBLE
	ui.connect("closed", func() -> void:
		tree.paused = was
		Input.mouse_mode = mm)
	return ui

func _ink(text: String, size: int, col := UiStyle.TEXT_INK) -> Label:
	var l := Label.new()
	l.text = text
	l.add_theme_font_size_override("font_size", size)
	l.add_theme_color_override("font_color", col)
	l.mouse_filter = Control.MOUSE_FILTER_IGNORE
	return l


func _btn(text: String, w := 150.0, primary := false, h := 62.0) -> Button:
	var b := Button.new()
	b.text = text
	b.custom_minimum_size = Vector2(w, h)
	b.focus_mode = Control.FOCUS_NONE
	b.add_theme_font_size_override("font_size", 24)
	if primary:
		UiStyle.style_primary(b)
	else:
		UiStyle.style_button(b, true)
	return b


func _say(text: String, col := UiStyle.TEXT_SOFT) -> void:
	_msg = text
	_msg_col = col


func _ok(text: String) -> void:
	_say(text, UiStyle.QING)


func _scroll_grid(cols: int, size: Vector2) -> Array:

	sc.custom_minimum_size = size
	sc.horizontal_scroll_mode = ScrollContainer.SCROLL_MODE_DISABLED

	g.columns = cols
	g.add_theme_constant_override("h_separation", int(GAP))
	g.add_theme_constant_override("v_separation", int(GAP))
	sc.add_child(g)
	return [sc, g]

## t：int = 老编号（0 仓库 / 1 坊市 / …，见 LEGACY_TABS）；String = 页签键（prep / stash / trader / shop / black / hero / rank）或中文页名
func _set_tab(t: Variant) -> void:
	var key := "prep"
	if t is int or t is float:
		key = str(LEGACY_TABS[clampi(int(t), 0, LEGACY_TABS.size() - 1)])
	elif t is String or t is StringName:
		var s := str(t)
		if TAB_KEYS.has(s):
			key = s
		elif TAB_NAMES.has(s):
			key = str(TAB_KEYS[TAB_NAMES.find(s)])
	_key = key
	_tab = LEGACY_TABS.find(key)
	_sel = {}
	_confirm = ""
	if _tab_btns.is_empty():
		return          # 还没 _ready：_ready 里会再翻一次
	for k in _tab_btns.size():
		_tab_btns[k].button_pressed = TAB_KEYS[k] == key
	_refresh()

## 面板居中；画面放不下（四周留一点边）就整块缩小，不放大
func _process(_dt: float) -> void:
	var vs := get_viewport_rect().size
	var ps := _panel.get_combined_minimum_size()
	_panel.size = ps
	var k := minf(1.0, minf((vs.x - 24.0) / ps.x, (vs.y - 16.0) / ps.y))
	_panel.scale = Vector2(k, k)
	_panel.position = ((vs - ps * k) * 0.5).floor()

func _list(src: String) -> Array:
	return ExProfile.stash() if src == "stash" else ExProfile.bag()


func _is_sel(src: String, i: int, slot := "") -> bool:
	return not _sel.is_empty() and str(_sel["src"]) == src and int(_sel["i"]) == i and str(_sel.get("slot", "")) == slot


func _select(src: String, i: int, slot := "") -> void:
	if _is_sel(src, i, slot):
		_sel = {}
	else:
		_sel = {"src": src, "i": i, "slot": slot}
	_msg = ""
	_confirm = ""
	_refresh()


func _sel_item() -> Dictionary:
	if _sel.is_empty():
		return {}
	var i := int(_sel["i"])
	match str(_sel["src"]):
		"bag", "stash":
			var l := _list(str(_sel["src"]))
			return l[i] if i >= 0 and i < l.size() else {}
		"equip":
			var eq := ExProfile.equipment()
			match str(_sel.get("slot", "")):
				"weapon":
					return eq["weapon"]
				"armor", "chip":
					var arr: Array = eq[str(_sel["slot"])]
					return arr[i] if i >= 0 and i < arr.size() else {}
		"shop":
			var goods := ExData.shop(_shop_cat)
			return ExData.make(str(goods[i]["id"])) if i >= 0 and i < goods.size() else {}
		"black":
			return ExData.make(str(ExData.BLACK[i]["id"])) if i >= 0 and i < ExData.BLACK.size() else {}
	return {}


## 一格方格（bag / stash 里的第 i 格）
func _grid_tile(src: String, i: int, size: Vector2, note := "") -> Button:
	var l := _list(src)
	var b := ExItemView.tile(l[i], size, _is_sel(src, i), "", note)
	b.pressed.connect(_select.bind(src, i))
	return b


func _fill_grid(g: GridContainer, src: String, cap: int, size: Vector2) -> void:
	var l := _list(src)
	for i in l.size():
		g.add_child(_grid_tile(src, i, size))
	for _i in maxi(cap - l.size(), 0):
		g.add_child(ExItemView.tile({}, size, false, "", "", ""))

## 英雄表：hero_defs.gd 有就用它（HeroDefs.ids() / HeroDefs.get(id)），没有就用 FALLBACK_HEROES
static func hero_ids() -> Array:
	return ExProfile.hero_ids()


static func hero_def(id: String) -> Dictionary:
	if ResourceLoader.exists(HERO_DEFS):
		var scr: GDScript = load(HERO_DEFS)
		if scr != null:
			for fn: String in ["def_of", "get_def", "def", "get_hero", "info"]:
				if scr.has_method(fn):
					var v: Variant = scr.call(fn, id)
					if v is Dictionary and not (v as Dictionary).is_empty():
						return v
			var consts := scr.get_script_constant_map()
			var tbl: Variant = consts.get("HEROES")
			if tbl is Dictionary and (tbl as Dictionary).has(id):
				return (tbl as Dictionary)[id]
			var v2: Variant = scr.call("get", id)
			if v2 is Dictionary and not (v2 as Dictionary).is_empty():
				return v2
	return FALLBACK_HEROES.get(id, {"name": id, "title": "", "desc": "", "skills": [], "awaken": ""})

## 卖出按钮：一个 / 整堆；品质 ≥5 第一次点只提示，第二次才卖
func _sell_acts(src: String, i: int, it: Dictionary) -> void:
	if not ExData.can_sell(it):
		return
	var n := maxi(int(it.get("n", 1)), 1)
	var opts: Array = [[-1, "卖出" if n == 1 else "全部卖出"]]
	if n > 1:
		opts.push_front([1, "卖 1 个"])
	for o: Array in opts:
		var cnt := int(o[0])
		var gain := ExData.unit_value(it) * (n if cnt < 0 else cnt)
		var key := "sell:%s:%d:%d" % [src, i, cnt]
		var label := "%s（%s）" % [str(o[1]), NumFmt.fmt(float(gain))]
		if _confirm == key:
			label = "确认卖出（%s）" % NumFmt.fmt(float(gain))
		_act(label, _confirm == key, true, func() -> String:
			if ExData.sell_confirm(it) and _confirm != key:
				_confirm = key
				_say("%s 是%s之物，再点一次确认卖出" % [str(it["name"]), ExData.q_name(int(it.get("q", 1)))], UiStyle.CINNABAR)
				return ""
			_confirm = ""
			var e := ExProfile.sell(src, i, cnt)
			if e == "":
				_ok("卖出 %s，得 %s 灵石" % [str(it["name"]), NumFmt.fmt(float(ExProfile.last_gain))])
			return e)


func _repair_act(src_or_slot: String, i: int, it: Dictionary) -> void:
	var cost := ExData.repair_cost(it)
	if cost <= 0:
		return
	_act("修理（%s）" % NumFmt.fmt(float(cost)), false, true, func() -> String:
		var e := ExProfile.repair(src_or_slot, i)
		if e == "":
			_ok("%s 修好了" % str(it["name"]))
		return e)


## 底栏一个按钮：fn 返回错误字符串（空 = 成功）。keep = 做完接着选同一格（还在的话）
func _act(text: String, primary: bool, keep: bool, fn: Callable) -> void:
	var b := _btn(text, maxf(150.0, 30.0 + 24.0 * text.length()), primary)
	b.pressed.connect(func() -> void:
		var sel := _sel.duplicate()
		var before := _sel_item()
		_msg = ""
		var err: String = fn.call()
		if err != "":
			_say(err, UiStyle.CINNABAR)
		_sel = {}
		if keep:
			# 还是同一样东西（一叠卖掉一个 / 买完接着买 / 等第二次确认）就接着选它
			_sel = sel
			var now := _sel_item()
			if now.is_empty() or str(now.get("id", "")) != str(before.get("id", "")):
				_sel = {}
		_refresh())
	_acts.add_child(b)


## 自检用：点正文里文字以 prefix 开头的按钮（扩容 / 一键卖 / 选用 / 坊市分类 …），返回点到没有
func press_body(prefix: String) -> bool:
	for n: Node in _body.find_children("*", "Button", true, false):
		var b := n as Button
		if b != null and not b.disabled and b.is_visible_in_tree() and not b.is_queued_for_deletion() and b.text.begins_with(prefix):
			b.pressed.emit()
			return true
	return false


## 自检用：点底栏里文字以 prefix 开头的按钮，返回点到没有
func press_action(prefix: String) -> bool:
	for c in _acts.get_children():
		if c is Button and not (c as Button).disabled and (c as Button).text.begins_with(prefix) and (c as CanvasItem).visible:
			(c as Button).pressed.emit()
			return true
	return false


## 自检用：底栏现在有哪些按钮
func action_texts() -> Array:
	var out: Array = []
	for c in _acts.get_children():
		if c is Button and (c as CanvasItem).visible:
			out.append((c as Button).text)
	return out


## 自检用：底栏那一行字
func detail_line() -> String:
	return _detail.text


## 入境页上那一行：这次带什么
static func summary() -> String:
	var eq := ExProfile.equipment()
	var w: Dictionary = eq["weapon"]
	var ar := 0
	for a: Dictionary in eq["armor"]:
		if not a.is_empty():
			ar += 1
	var ch := 0
	for c: Dictionary in eq["chip"]:
		if not c.is_empty():
			ch += 1
	var ws := "%s·%s" % [str(w["name"]), ExData.q_name(int(w.get("q", 1)))] if not w.is_empty() else "空手"
	return "%s　法衣 %d/4　法宝 %d/3　背包 %d/%d 格　%s 灵石" % [ws, ar, ch, ExProfile.bag().size(), ExProfile.BAG_CAP,
			NumFmt.fmt(float(ExProfile.lingshi()))]
