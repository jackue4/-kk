class MockMgr extends Node3D:
	var level: Node3D
	var player: Node3D
	var cam: Node3D
	var net: Node = null
	var map_id := "desert"
	var font: Font
	var mobile := false
	var hud: CanvasLayer
	var time_left := 872.0
	var bag: Array = []
	var equip: Dictionary = ExData.empty_equip()
	var channel: Dictionary = {}
	var cards: Array = []
	var downed := false
	var downed_left := 20.0
	var can_self_revive := false
	var kills := 0
	var pk_kills := 0
	var doctor_state := ""
	var result: Dictionary = {}
	var calls_log: Array = []

	func carry_value() -> int:
		return ExData.extract_value(bag)

	func _log(s: String) -> void:
		calls_log.append(s)
		print("[exhud] mgr.", s)

	func ui_card(key: String) -> void:
		_log("ui_card(%s)" % key)

	func ui_use(i: int) -> void:
		_log("ui_use(%d)" % i)
		if i >= 0 and i < bag.size():
			var it: Dictionary = bag[i]
			it["n"] = int(it.get("n", 1)) - 1
			if int(it["n"]) <= 0:
				bag.remove_at(i)

	func ui_wear(i: int) -> void:
		_log("ui_wear(%d)" % i)
		if i < 0 or i >= bag.size():
			return
		var it: Dictionary = bag[i]
		var so: Array = ExData.slot_of(it)
		if so.is_empty():
			return
		bag.remove_at(i)
		var old: Dictionary = {}
		match str(so[0]):
			"weapon":
				old = equip["weapon"]
				equip["weapon"] = it
			"armor":
				old = equip["armor"][int(so[1])]
				equip["armor"][int(so[1])] = it
			"chip":
				var chips: Array = equip["chip"]
				var at := chips.find({})
				if at < 0:
					at = 0
				old = chips[at]
				chips[at] = it
		if not old.is_empty():
			bag.append(old)

	func ui_unequip(slot: String, idx: int) -> void:
		_log("ui_unequip(%s,%d)" % [slot, idx])
		var it: Dictionary = equip["weapon"] if slot == "weapon" else equip[slot][idx]
		if it.is_empty():
			return
		bag.append(it)
		if slot == "weapon":
			equip["weapon"] = {}
		else:
			equip[slot][idx] = {}

	func ui_drop(i: int, n: int = -1) -> void:
		_log("ui_drop(%d,%d)" % [i, n])
		if i >= 0 and i < bag.size():
			bag.remove_at(i)

	func ui_self_revive() -> void:
		_log("ui_self_revive()")

	func ui_suicide() -> void:
		_log("ui_suicide()")

	func ui_result_done() -> void:
		_log("ui_result_done()")

func _initialize() -> void:
	var args := OS.get_cmdline_user_args()
	_touch = args.has("--touch")
	var compat := str(RenderingServer.get_current_rendering_method()) == "gl_compatibility"
	UiStyle.force_bundled = compat or args.has("--web-fonts")
	_tag = "exhud" + ("_gl" if compat else "") + ("_touch" if _touch else "")
	DirAccess.make_dir_recursive_absolute(ProjectSettings.globalize_path("res://_shots/preview"))
	for a: Array in [["move_forward", KEY_W], ["move_back", KEY_S], ["move_left", KEY_A], ["move_right", KEY_D], ["attack", KEY_J],
			["guard", KEY_K], ["sprint", KEY_SHIFT], ["air_shot", KEY_E], ["bag", KEY_B]]:
		if not InputMap.has_action(str(a[0])):
			InputMap.add_action(str(a[0]))
			var ev := InputEventKey.new()
			ev.physical_keycode = a[1]
			InputMap.action_add_event(str(a[0]), ev)
	_font = UiStyle.kai()
	_build_scene()
	_run()

## 英雄档的触屏按钮：[圆心, 半径, 字, 只在空中?]（境± 只在大厅，不画）
func _buttons() -> Array:
	var vs := _vs()
	var out: Array = []
	var hero_txt := {"shot": "瞬", "art": "回", "summon": "月", "ult": "刃"}
	for b: Dictionary in TC.BUTTONS:
		var id := str(b["id"])
		if id == "rp" or id == "rn":
			continue
		var off: Vector2 = b["off"]
		var r := float(b["r"])
		if TC.HERO_POS.has(id):
			off = TC.HERO_POS[id][0]
			r = float(TC.HERO_POS[id][1])
		var c := Vector2(vs.x + off.x, off.y) if str(b["anchor"]) == "tr" else vs + off
		out.append([c, r, str(hero_txt.get(id, b["text"])), TC.HERO_SLOT.has(id)])
	return out


func _draw_pad() -> void:
	var vs := _vs()
	var bf := UiStyle.brush()
	var ring := UiStyle.tex("enso")
	var items: Array = [[Vector2(210.0, vs.y - 210.0), 110.0, "", false]] + _buttons()
	for it: Array in items:
		var c: Vector2 = it[0]
		var r: float = it[1]
		_pad.draw_circle(c, r * 0.97, Color(UiStyle.PAPER, 0.86 if str(it[2]) != "" else 0.4), true, -1.0, true)
		if ring != null:
			var er := r * 1.19
			_pad.draw_texture_rect(ring, Rect2(c - Vector2(er, er), Vector2(er, er) * 2.0), false, Color(UiStyle.LINE, 0.9))
		var txt: String = it[2]
		if txt != "":
			var fs := int(r * 0.7)
			var ts := bf.get_string_size(txt, HORIZONTAL_ALIGNMENT_LEFT, -1, fs)
			_pad.draw_string(bf, c + Vector2(-ts.x * 0.5, ts.y * 0.3), txt, HORIZONTAL_ALIGNMENT_LEFT, -1, fs, UiStyle.TEXT_INK)

func _shot(state: String) -> void:
	_pad.visible = _touch and not bool(_hud.call("bag_open")) if is_instance_valid(_hud) else _touch     # 游戏里背包 / 结算开着时触屏层不画

	var img := root.get_texture().get_image()
	var tag := "%s_%s" % [_tag, state]
	img.save_png("res://_shots/%s.png" % tag)
	var pv := img.duplicate() as Image
	pv.convert(Image.FORMAT_RGB8)
	pv.resize(800, int(800.0 * img.get_height() / img.get_width()), Image.INTERPOLATE_BILINEAR)
	pv.save_jpg("res://_shots/preview/%s.jpg" % tag, 0.88)
	print("[exhud] shot _shots/preview/%s.jpg" % tag)


## 原分辨率截一块（units = 界面单位的矩形），放大 1.5 倍存 jpg：看字有没有缺字方块
func _zoom(state: String, units: Rect2) -> void:
	await RenderingServer.frame_post_draw
	var img := root.get_texture().get_image()
	var k := float(img.get_width()) / _vs().x
	var r := Rect2i(Vector2i(units.position * k), Vector2i(units.size * k)).intersection(Rect2i(Vector2i.ZERO, img.get_size()))
	var sub := img.get_region(r)
	sub.convert(Image.FORMAT_RGB8)
	sub.resize(int(r.size.x * 1.5), int(r.size.y * 1.5), Image.INTERPOLATE_BILINEAR)
	sub.save_jpg("res://_shots/preview/%s_%s_zoom.jpg" % [_tag, state], 0.9)
	print("[exhud] zoom _shots/preview/%s_%s_zoom.jpg" % [_tag, state])

## 界面块和触屏按钮 / 摇杆 / 位置框的重叠检查（ground_only = 这块只在地面出现，只和地面按钮比）
func _no_overlap(name: String, r: Rect2, ground_only := false, with_tgt := true) -> void:
	var bad: Array = []
	if _touch:
		var items: Array = [[Vector2(210.0, _vs().y - 210.0), 110.0, "摇杆", false]] + _buttons()
		for it: Array in items:
			if ground_only and bool(it[3]):
				continue
			var c: Vector2 = it[0]
			var rr: float = it[1]
			var q := Vector2(clampf(c.x, r.position.x, r.end.x), clampf(c.y, r.position.y, r.end.y))
			if q.distance_to(c) < rr:
				bad.append(str(it[2]))
	else:
		if r.intersects(_skill_rect()) and not ground_only:
			bad.append("技能栏")
	if with_tgt and r.intersects(_tgt_rect()):
		bad.append("目标面板")
	var plate_r := Rect2(_plate.position, _plate.size * _plate.scale)
	if r.intersects(plate_r):
		bad.append("玩家牌")
	_check("不重叠：" + name, bad.is_empty(), "%s %s" % [r, ",".join(bad)])
