static var _poses: Dictionary = {}
static var _seqs: Dictionary = {}


static func get_pose(name: String) -> Dictionary:
	if not _poses.has(name):
		_poses[name] = build(DEF[name])
	return _poses[name]


## 时间轴：返回 [[t, pose_dict, ease], ...]
static func seq(name: String) -> Array:
	if not _seqs.has(name):
		var out: Array = []
		for k: Array in SEQ[name]:
			out.append([float(k[0]), get_pose(k[1]), int(k[2])])
		_seqs[name] = out
	return _seqs[name]


static func build(d: Dictionary) -> Dictionary:
	var ar: Vector3 = (d.get("ar", Vector3(0.1, -1.0, 0.0)) as Vector3).normalized()
	var bl: Vector3 = (d.get("bl", Vector3(0.0, 0.0, -1.0)) as Vector3).normalized()
	var al: Vector3 = (d.get("al", Vector3(-0.1, -1.0, 0.0)) as Vector3).normalized()
	var out := {"ar": arm_quat(ar, bl), "bl": blade_quat(bl, ar), "al": arm_quat(al, d.get("alt", Vector3(0.0, 0.0, -1.0)) as Vector3)}
	for k: String in FLOATS:
		out[k] = float(d.get(k, DEFAULTS.get(k, 0.0)))
	return out


## 手臂朝 dir 伸出（模型本地 -Y 指向 dir），手腕一侧（本地 -Z）尽量朝向 toward
static func arm_quat(dir: Vector3, toward: Vector3) -> Quaternion:
	var y := -dir
	var z := -toward
	z -= y * z.dot(y)
	if z.length_squared() < 1e-4:
		z = Vector3.BACK - y * y.z
		if z.length_squared() < 1e-4:
			z = Vector3.UP - y * y.y
	z = z.normalized()
	return Basis(y.cross(z), y, z).get_rotation_quaternion()


## 剑尖（模型 +Y）朝 bl，剑面法线偏向手臂一侧
static func blade_quat(bl: Vector3, ar: Vector3) -> Quaternion:
	var y := bl
	var z := ar - y * ar.dot(y)
	if z.length_squared() < 1e-4:
		z = Vector3.BACK - y * y.z
		if z.length_squared() < 1e-4:
			z = Vector3.UP - y * y.y
	z = z.normalized()
	return Basis(y.cross(z), y, z).get_rotation_quaternion()


static func lerp_pose(a: Dictionary, b: Dictionary, u: float) -> Dictionary:
	var out := {
		"ar": (a["ar"] as Quaternion).slerp(b["ar"], u),
		"bl": (a["bl"] as Quaternion).slerp(b["bl"], u),
		"al": (a["al"] as Quaternion).slerp(b["al"], u),
	}
	for k: String in FLOATS:
		out[k] = lerpf(a[k], b[k], u)
	return out
