## id 写法：售卖物 s01..s93 / sx1..sx5、命牌 tag、兵刃 w_<家族>_<品质>（w_qingfeng_5）或独特兵刃 w_u_*、
## 法衣 a_<部位 0..3>_<品质>、法宝 c_<regen|shell|pierce>_<品质 4..8>、药 med_*、revive、符 thr_*、key、ward、voucher、材料 m_*；
## "@w5" = 史诗档随机家族的兵刃（掉落池里用）
static func make(id: String, n := 1) -> Dictionary:
	if id.begins_with("@w"):
		var fq := clampi(int(id.substr(2)), 1, 7)
		return make("w_%s_%d" % [FAMILIES[randi() % FAMILIES.size()], fq])
	if SELL.has(id):
		var s: Dictionary = SELL[id]
		var it := _base(id, "sell", str(s["name"]), int(s["q"]), int(s["v"]), mini(n, STACK_SELL))
		it["da_name"] = str(s["da"])
		it["cat"] = str(s["cat"])
		it["label"] = str(s["tag"])
		return it
	if MATS.has(id):
		var m: Dictionary = MATS[id]
		var mt := _base(id, "mat", str(m["name"]), int(m["q"]), int(m["v"]), mini(n, STACK))
		mt["da_name"] = str(m["da"])
		return mt
	if UNIQUE.has(id):
		var u: Dictionary = UNIQUE[id]
		var uw := _base(id, "weapon", str(u["name"]), int(u["q"]), int(u["v"]), 1)
		_weapon_stats(uw, str(u["fam"]))
		uw["unique"] = true
		uw["da_name"] = str(u["da"])
		return uw
	if MED_NAMES.has(id):
		var md: Dictionary = MED_DA.get(id, {})
		var it2 := _base(id, "med", str(MED_NAMES[id]), int(md.get("q", 1)), int(md.get("v", 0)), mini(n, STACK))
		var hot: Array = MED_HOT_DA.get(id, [0.0, 0.0])
		it2["heal"] = int(roundf(float(md.get("da_heal", 0)) * HEAL_K))
		it2["hot"] = float(hot[0]) * HEAL_K
		it2["dur"] = float(hot[1])
		it2["use_t"] = float(md.get("use_t", 3.0))
		return it2
	if THROW.has(id):
		var td: Dictionary = THROW[id]
		var da: Dictionary = THROW_DA.get(id, {})
		var it3 := _base(id, "throw", str(td["name"]), int(da.get("q", 2)), int(da.get("v", 0)), mini(n, STACK))
		it3["fx"] = str(td["fx"])
		it3["dmg"] = int(td["dmg"])
		it3["r"] = float(td["r"])
		it3["t"] = float(td["t"])
		it3["dps"] = float(td["dps"])
		return it3
	match id:
		"tag":
			var tg := _base("tag", "tag", str(TAG["name"]), int(TAG["q"]), int(TAG["v"]), 1)
			tg["owner"] = ""
			return tg
		"revive":
			var rv := _base("revive", "revive", str(REVIVE["name"]), int(REVIVE["q"]), int(REVIVE["v"]), mini(n, STACK))
			rv["use_t"] = float(REVIVE["use_t"])
			return rv
		"key":
			return _base("key", "key", str(KEY["name"]), int(KEY["q"]), 0, mini(n, STACK))
		"ward":
			return _base("ward", "ward", str(WARD["name"]), int(WARD["q"]), 0, mini(n, STACK))
		"voucher":
			return _base("voucher", "voucher", str(VOUCHER["name"]), int(VOUCHER["q"]), 0, n)
	var p := id.split("_")
	if p.size() == 3 and p[0] == "w" and FAMILIES.has(p[1]):
		var wq := clampi(int(p[2]), 1, 7)
		var w := _base(id, "weapon", str(FAMILY_NAMES[p[1]]), wq, int(WEAPON_VALUE[wq]), 1)
		_weapon_stats(w, p[1])
		return w
	if p.size() == 3 and p[0] == "a":
		var slot := clampi(int(p[1]), 0, 3)
		var aq := clampi(int(p[2]), 1, 7)
		var a := _base("a_%d_%d" % [slot, aq], "armor", "%s%s" % [ARMOR_SETS[aq], ARMOR_PARTS[slot]], aq, int(ARMOR_VALUE[slot][aq]), 1)
		a["slot"] = slot
		a["dr_pve"] = float(ARMOR_SET_PVE[aq]) / 4.0
		a["dr_pvp"] = float(ARMOR_SET_PVP[aq]) / 4.0
		return a
	if p.size() == 3 and p[0] == "c" and CHIP_FX.has(p[1]):
		var cq := clampi(int(p[2]), 4, 8)
		var c := _base("c_%s_%d" % [p[1], cq], "chip", str(CHIP_NAMES[p[1]]), cq, int(CHIP_BASE[cq]), 1)
		c["fx"] = p[1]
		c["base"] = int(CHIP_BASE[cq])
		c["dur_max"] = int(CHIP_DUR[cq])
		c["dur"] = int(CHIP_DUR[cq])
		_chip_stats(c)
		return c
	return {}

## JSON 读回来的物品：数字转回 int、数值字段按表重算（改平衡不用迁移存档）。认不出的 id 原样返回（只修类型）
static func fix_item(it: Dictionary) -> Dictionary:
	var id := str(it.get("id", ""))
	var n := maxi(int(it.get("n", 1)), 1)
	var f := make(id, n) if not id.begins_with("@") else {}
	if f.is_empty():
		for k: String in ["q", "value", "n", "dur", "dur_max", "base", "slot", "heal", "dmg"]:
			if it.has(k) and (it[k] is float or it[k] is int):
				it[k] = int(it[k])
		return it
	f["n"] = n if stacks(f) else 1
	match str(f["kind"]):
		"chip":
			f["dur"] = clampi(int(it.get("dur", f["dur_max"])), 0, int(f["dur_max"]))
		"tag":
			f = make_tag(str(it.get("owner", "")))
		"weapon":
			if it.has("fam") and not bool(f.get("unique", false)):
				_weapon_stats(f, str(it["fam"]))
	return f

## 单件回收价（法宝按耐久折：基础价 × 当前耐久 / 上限）
static func unit_value(it: Dictionary) -> int:
	if it.is_empty():
		return 0
	if str(it.get("kind", "")) == "chip":
		var mx := maxi(int(it.get("dur_max", 1)), 1)
		return int(floorf(float(it.get("base", it.get("value", 0))) * float(clampi(int(it.get("dur", mx)), 0, mx)) / float(mx)))
	return int(it.get("value", 0))


## 这一格的回收价合计（单价 × 堆叠数）。注意：是合计，不用再乘 n
static func value_of(it: Dictionary) -> int:
	return unit_value(it) * maxi(int(it.get("n", 1)), 1)


## 售卖物（SPEC 3.3：sell + 命牌）：禁止带进禁地、带出时计入带出价值 / 段位
static func is_sellable(it: Dictionary) -> bool:
	var k := str(it.get("kind", ""))
	return k == "sell" or k == "tag"


## 商人收不收（有回收价、不是天启）
static func can_sell(it: Dictionary) -> bool:
	return not it.is_empty() and unit_value(it) > 0 and int(it.get("q", 1)) < NO_SELL_Q


## 卖之前要不要二次确认（品质 ≥5）
static func sell_confirm(it: Dictionary) -> bool:
	return int(it.get("q", 1)) >= SELL_CONFIRM_Q


## 一列格子里售卖物的价值合计（带出价值 / 背包价值）
static func extract_value(list: Array) -> int:
	var s := 0
	for it: Variant in list:
		if it is Dictionary and is_sellable(it):
			s += value_of(it)
	return s


## 法宝修理费：基础价 × 缺损比例 × 0.5
static func repair_cost(it: Dictionary) -> int:
	if str(it.get("kind", "")) != "chip":
		return 0
	var mx := maxi(int(it.get("dur_max", 1)), 1)
	var miss := 1.0 - float(clampi(int(it.get("dur", mx)), 0, mx)) / float(mx)
	return int(ceilf(float(it.get("base", 0)) * miss * REPAIR_K))

static func stacks(it: Dictionary) -> bool:
	return STACK_KINDS.has(str(it.get("kind", "")))


static func stack_max(it: Dictionary) -> int:
	if not stacks(it):
		return 1
	return STACK_SELL if str(it.get("kind", "")) == "sell" else STACK


## 两格能不能叠成一格：同 id、都是可叠的
static func same_stack(a: Dictionary, b: Dictionary) -> bool:
	return stacks(a) and str(a.get("id", "")) == str(b.get("id", "")) and str(a.get("kind", "")) == str(b.get("kind", ""))


## 一列格子里还能放进 it 多少个（先叠进已有的同 id 格、再占空格；不改 list）
static func room_for(list: Array, it: Dictionary, cap: int) -> int:
	if it.is_empty():
		return 0
	var mx := stack_max(it)
	var room := 0
	if stacks(it):
		for s: Variant in list:
			if s is Dictionary and same_stack(s, it):
				room += maxi(mx - int((s as Dictionary).get("n", 1)), 0)
	room += maxi(cap - list.size(), 0) * mx
	return room


## 把 it（整堆）放进 list：先叠进同 id 的格子、再占空格，最多 cap 格。返回放不下剩下的个数（0 = 全放进去了）。
## it 本身不改（放进去的是拷贝）
static func add_to(list: Array, it: Dictionary, cap: int) -> int:
	if it.is_empty():
		return 0
	var left := maxi(int(it.get("n", 1)), 1)
	if not stacks(it):
		while left > 0 and list.size() < cap:
			var c := it.duplicate(true)
			c["n"] = 1
			list.append(c)
			left -= 1
		return left
	var mx := stack_max(it)
	for s: Variant in list:
		if left <= 0:
			break
		if s is Dictionary and same_stack(s, it):
			var sd: Dictionary = s
			var k := mini(maxi(mx - int(sd.get("n", 1)), 0), left)
			sd["n"] = int(sd.get("n", 1)) + k
			left -= k
	while left > 0 and list.size() < cap:
		var c2 := it.duplicate(true)
		var k2 := mini(mx, left)
		c2["n"] = k2
		list.append(c2)
		left -= k2
	return left


## list 里一共有几个 id
static func count_id(list: Array, id: String) -> int:
	var s := 0
	for it: Variant in list:
		if it is Dictionary and str((it as Dictionary).get("id", "")) == id:
			s += maxi(int((it as Dictionary).get("n", 1)), 1)
	return s


## 从 list 里拿掉 n 个 id（从后往前扣，空了的格子删掉），返回实际拿掉几个
static func take_id(list: Array, id: String, n := 1) -> int:
	var got := 0
	var i := list.size() - 1
	while i >= 0 and got < n:
		var it: Variant = list[i]
		if it is Dictionary and str((it as Dictionary).get("id", "")) == id:
			var d: Dictionary = it
			var k := mini(maxi(int(d.get("n", 1)), 1), n - got)
			d["n"] = int(d.get("n", 1)) - k
			got += k
			if int(d["n"]) <= 0:
				list.remove_at(i)
		i -= 1
	return got


## 把同 id 的散堆合并（掉落 roll 出来是一件一件的，给 UI / 结算用）
static func merge(list: Array) -> Array:
	var out: Array = []
	for it: Variant in list:
		if it is Dictionary and not (it as Dictionary).is_empty():
			add_to(out, it, 99999)
	return out

## 装备栏补齐结构（读档 / 联机收到的半截数据）
static func fix_equip(e: Variant) -> Dictionary:
	var out := empty_equip()
	if not (e is Dictionary):
		return out
	var d: Dictionary = e
	if d.get("weapon") is Dictionary and not (d["weapon"] as Dictionary).is_empty():
		out["weapon"] = fix_item(d["weapon"])
	for key: String in ["armor", "chip"]:
		var src: Variant = d.get(key)
		if src is Array:
			var arr: Array = out[key]
			for i in mini((src as Array).size(), arr.size()):
				var x: Variant = (src as Array)[i]
				if x is Dictionary and not (x as Dictionary).is_empty():
					arr[i] = fix_item(x)
	return out

	for key: String in ["armor", "chip"]:
		var arr: Variant = equip.get(key, [])
		if arr is Array:
			for x: Variant in arr:
				if x is Dictionary and not (x as Dictionary).is_empty():

	return out


## 兵刃出手倍率（it = 兵刃字典，空 = 空手 1.0）

## 法宝合计：{regen 每秒回血, shell 护体值, pierce 伤害加成（0..1）}；耐久 0 的不算
static func chip_fx(equip: Dictionary, pvp: bool) -> Dictionary:
	var fx := {"regen": 0.0, "shell": 0.0, "pierce": 0.0}
	var arr: Variant = equip.get("chip", [])
	if arr is Array:
		for x: Variant in arr:
			if not (x is Dictionary) or (x as Dictionary).is_empty():
				continue
			var c: Dictionary = x
			if int(c.get("dur", 0)) <= 0:
				continue
			var q := clampi(int(c.get("q", 4)), 4, 8)
			match str(c.get("fx", "")):
				"regen":
					fx["regen"] = float(fx["regen"]) + float(CHIP_REGEN[q])
				"shell":
					fx["shell"] = float(fx["shell"]) + float(CHIP_SHELL[q])
				"pierce":
					fx["pierce"] = float(fx["pierce"]) + float(CHIP_PIERCE_PVP[q] if pvp else CHIP_PIERCE_PVE[q])
	return fx


## 出手倍率合计 = 兵刃 × (1 + 破军印)；PvP 夹到 PVP_ATK_CAP
static func atk_mult(equip: Dictionary, pvp: bool) -> float:
	var k := weapon_mult(equip.get("weapon", {}), pvp) * (1.0 + float(chip_fx(equip, pvp)["pierce"]))
	return minf(k, PVP_ATK_CAP) if pvp else k


## 受到真人伤害 bar_dmg（条上点）：每件还有耐久的法宝扣 bar_dmg × CHIP_WEAR，返回这次耗尽的法宝名字
static func wear_chips(equip: Dictionary, bar_dmg: float) -> Array:
	var broke: Array = []
	var arr: Variant = equip.get("chip", [])
	if not (arr is Array) or bar_dmg <= 0.0:
		return broke
	for x: Variant in arr:
		if x is Dictionary and not (x as Dictionary).is_empty():
			var c: Dictionary = x
			var d0 := int(c.get("dur", 0))
			if d0 <= 0:
				continue
			c["dur"] = maxi(d0 - int(ceilf(bar_dmg * float(CHIP_WEAR))), 0)
			if int(c["dur"]) == 0:
				broke.append(str(c.get("name", "")))
	return broke


## 陨落掉落（SPEC 4.6 第 2、3 条）：直接改 bag / equip，返回 {drop 全部掉的, sell 掉的售卖物堆, chips 掉的法宝, ward 用掉护宝符没有}。
##   售卖物：从背包里随机掉 stacks 堆（< 0 = 全掉）；背包里有护宝符时先把价值最高的 WARD_KEEP 堆剔出掉落池（≤WARD_KEEP 堆就不掉），并用掉一张
##   法宝：全掉（带当前耐久）；兵刃 / 法衣 / 药 / 符 / 钥匙不掉
## E 的死亡流程和 ex_profile 的中途离线结算共用这一份
static func death_drops(bag: Array, equip: Dictionary, rng: RandomNumberGenerator, stacks := DEATH_DROP_STACKS) -> Dictionary:
	var pool: Array = []
	for i in bag.size():
		if bag[i] is Dictionary and is_sellable(bag[i]):
			pool.append(i)
	var ward := false
	if not pool.is_empty() and count_id(bag, "ward") > 0:
		pool.sort_custom(func(a: int, b: int) -> bool: return value_of(bag[a]) > value_of(bag[b]))
		pool = pool.slice(WARD_KEEP)
		ward = true
	var pick: Array = []
	if stacks < 0:
		pick = pool.duplicate()
	else:
		var left := pool.duplicate()
		while pick.size() < stacks and not left.is_empty():
			pick.append(left.pop_at(rng.randi() % left.size()))
	pick.sort()
	var sold: Array = []
	for k in range(pick.size() - 1, -1, -1):
		sold.push_front(bag[int(pick[k])])
		bag.remove_at(int(pick[k]))
	if ward:
		take_id(bag, "ward", 1)
	var chips: Array = []
	var arr: Variant = equip.get("chip", [])
	if arr is Array:
		for i in (arr as Array).size():
			var c: Variant = (arr as Array)[i]
			if c is Dictionary and not (c as Dictionary).is_empty():
				chips.append(c)
				(arr as Array)[i] = {}
	return {"drop": sold + chips, "sell": sold, "chips": chips, "ward": ward}


## 能不能用（局内背包「使用」）：药 / 强心丹 / 符
static func usable(it: Dictionary) -> bool:

## 能不能穿戴：兵刃 / 法衣 / 法宝
static func wearable(it: Dictionary) -> bool:

## 一件东西在装备栏里的位置：["weapon", 0] / ["armor", 部位] / ["chip", -1（找空位）] / [] 不能穿
static func slot_of(it: Dictionary) -> Array:

		"weapon":
			return ["weapon", 0]
		"armor":
			return ["armor", clampi(int(it.get("slot", 0)), 0, 3)]
		"chip":
			return ["chip", -1]
	return []

## 抽一张表（SPEC 3.6）：每抽按权重选组（DarkArea：rand = random(1, 总权重)）、组内等概率、有放回；draws < 0 用表里的抽数。
## 返回一件一件的物品（n = 1，不合并；要合并用 merge）。另按 KEY_P / KEY_P_BOSS 独立掷一次禁室玉钥
static func roll(table: String, rng: RandomNumberGenerator, draws := -1) -> Array[Dictionary]:
	var out: Array[Dictionary] = []
	if not TABLES.has(table):
		push_warning("[ExData] 没有掉落表 %s" % table)
		return out
	var t: Dictionary = TABLES[table]
	var groups: Array = t["groups"]
	var total := 0
	for g: Array in groups:
		total += int(g[0])
	var nd := int(t["draws"]) if draws < 0 else draws
	for _i in nd:
		var r := rng.randi_range(1, maxi(total, 1))
		var pick: Array = groups[groups.size() - 1]
		for g: Array in groups:
			r -= int(g[0])
			if r <= 0:
				pick = g
				break
		var ids: Array = POOLS.get(pick[1], []) if pick[1] is String else pick[1]
		if ids.is_empty():
			continue
		var it := _make_rng(str(ids[rng.randi() % ids.size()]), rng)
		if not it.is_empty():
			out.append(it)
	var kp := key_p(table)
	if kp > 0.0 and rng.randf() < kp:
		out.append(make("key"))
	return out


## 这张表另外掉禁室玉钥的概率（开一次箱 / 杀一次首领）
static func key_p(table: String) -> float:
	if KEY_P.has(table):
		return float(KEY_P[table])
	if table.contains("Boss") or table.begins_with("ghost"):
		return KEY_P_BOSS
	return 0.0


static func _make_rng(id: String, rng: RandomNumberGenerator) -> Dictionary:
	if id.begins_with("@w"):
		var q := clampi(int(id.substr(2)), 1, 7)
		return make("w_%s_%d" % [FAMILIES[rng.randi() % FAMILIES.size()], q])
	return make(id)


## 箱子用哪张表：map_id "desert" / "lab"（其余 = 市中心那套 junior*），box_kind "weapon" / "supply" / "med"，locked = 钥匙房里的
static func box_table(map_id: String, box_kind: String, locked := false) -> String:
	match map_id:
		"desert":
			if locked:
				return {"weapon": "desertLockedSWeaponBox", "supply": "desertLockedSupplyBox"}.get(box_kind, "seniorMedicalBox")
			return {"weapon": "seniorWeaponBox", "supply": "seniorSupplyBox"}.get(box_kind, "seniorMedicalBox")
		"lab":
			if locked:
				return {"weapon": "labLockedSWeaponBox", "supply": "labLockedSupplyBox"}.get(box_kind, "labMedicalBox")
			return {"weapon": "labWeaponBox", "supply": "labSupplyBox"}.get(box_kind, "labMedicalBox")
	return {"weapon": "juniorWeaponBox", "supply": "juniorSupplyBox"}.get(box_kind, "juniorMedicalBox")

## 坊市一类货架：[{id, price}]（price = 单价，灵石）。顺序固定
static func shop(cat: String) -> Array:
	if _shop_cache.has(cat):
		return _shop_cache[cat]
	var out: Array = []
	match cat:
		"weapon":
			for q in range(1, SHOP_MAX_Q + 1):
				for fam: String in FAMILIES:
					out.append({"id": "w_%s_%d" % [fam, q], "price": int(WEAPON_SHOP[q])})
		"armor":
			for q in range(1, SHOP_MAX_Q + 1):
				for s in 4:
					out.append({"id": "a_%d_%d" % [s, q], "price": int(ARMOR_SHOP[s][q])})
		"med":
			for id: String in MED_ORDER:
				out.append({"id": id, "price": int((MED_DA.get(id, {}) as Dictionary).get("price", 0))})
		"throw":
			for id: String in THROW_ORDER:
				out.append({"id": id, "price": int((THROW_DA.get(id, {}) as Dictionary).get("price", 0))})
		"chip":
			for fx: String in CHIP_FX:
				for q: int in CHIP_Q:
					out.append({"id": "c_%s_%d" % [fx, q], "price": int(CHIP_BASE[q]) * CHIP_SHOP_K})
	out = out.filter(func(g: Dictionary) -> bool: return int(g["price"]) > 0)
	_shop_cache[cat] = out
	return out


## 坊市单价（0 = 坊市不卖）
static func shop_price(id: String) -> int:
	for cat: String in SHOP_CATS:
		for g: Dictionary in shop(cat):
			if str(g["id"]) == id:
				return int(g["price"])
	return 0


## 黑市单价（仙玉；0 = 不卖）
static func black_price(id: String) -> int:
	for g: Dictionary in BLACK:
		if str(g["id"]) == id:
			return int(g["price"])
	return 0


## 仓库 level 级（1..14）的格数
static func wh_cap(level: int) -> int:
	return int(WAREHOUSE[clampi(level, 1, WAREHOUSE.size()) - 1][0])


## 从 level 升到下一级要什么：[灵石, 仓库凭证张数]；满级返回 []
static func wh_next_cost(level: int) -> Array:
	if level >= WAREHOUSE.size():
		return []
	var row: Array = WAREHOUSE[level]
	return [int(row[1]), int(row[2])]


## 累计带出价值 → 段位下标（0 = 未入流 … 13 = 化神）
static func rank_of(total: int) -> int:
	var r := 0
	for i in RANK_NEED.size():
		if total >= int(RANK_NEED[i]):
			r = i
	return r


static func rank_name(i: int) -> String:
	return str(RANK_NAMES[clampi(i, 0, RANK_NAMES.size() - 1)])


## 下一档要多少（满级返回 -1）
static func rank_next_need(i: int) -> int:
	return int(RANK_NEED[i + 1]) if i + 1 < RANK_NEED.size() else -1
