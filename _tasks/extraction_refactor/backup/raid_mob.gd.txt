## 返回 [身体网格, 肢体 A 网格, 肢体 B 网格, 肢体轴心 A, 肢体轴心 B]
static func _meshes(d: Dictionary) -> Array:
	var key: String = d["name"]
	if _mesh_cache.has(key):
		return _mesh_cache[key]
	var cols: Array = d["cols"]
	var a: Color = cols[0]
	var b: Color = cols[1]
	var e: Color = cols[2]
	e = Color(e.r * 1.8, e.g * 1.8, e.b * 1.8)
	var body := SurfaceTool.new()
	var la := SurfaceTool.new()
	var lb := SurfaceTool.new()
	for st: SurfaceTool in [body, la, lb]:
		st.begin(Mesh.PRIMITIVE_TRIANGLES)
	var pa := Vector3(0, 0.6, 0)
	var pb := Vector3(0, 0.6, 0)
	match str(d["shape"]):
		"beast":
			Data.add_box(body, Vector3(0.7, 0.6, 1.5), _at(Vector3(0, 0.85, 0)), a)
			Data.add_box(body, Vector3(0.6, 0.55, 0.6), _at(Vector3(0, 1.1, -0.95)), a)
			Data.add_box(body, Vector3(0.4, 0.3, 0.4), _at(Vector3(0, 0.98, -1.35)), b)
			Data.add_box(body, Vector3(0.16, 0.3, 0.12), _at(Vector3(-0.2, 1.5, -0.85)), b)
			Data.add_box(body, Vector3(0.16, 0.3, 0.12), _at(Vector3(0.2, 1.5, -0.85)), b)
			Data.add_box(body, Vector3(0.12, 0.1, 0.06), _at(Vector3(-0.17, 1.2, -1.26)), e)
			Data.add_box(body, Vector3(0.12, 0.1, 0.06), _at(Vector3(0.17, 1.2, -1.26)), e)
			Data.add_box(body, Vector3(0.18, 0.18, 0.8), _at(Vector3(0, 1.0, 1.05)), b)
			# 对角的两条腿一组：一组前摆时另一组后摆
			Data.add_box(la, Vector3(0.2, 0.6, 0.22), _at(Vector3(-0.28, -0.3, -0.5)), b)
			Data.add_box(la, Vector3(0.2, 0.6, 0.22), _at(Vector3(0.28, -0.3, 0.5)), b)
			Data.add_box(lb, Vector3(0.2, 0.6, 0.22), _at(Vector3(0.28, -0.3, -0.5)), b)
			Data.add_box(lb, Vector3(0.2, 0.6, 0.22), _at(Vector3(-0.28, -0.3, 0.5)), b)
		"floater":
			Data.add_box(body, Vector3(0.8, 0.8, 0.8), _at(Vector3(0, 1.0, 0)), a)
			Data.add_box(body, Vector3(0.2, 0.16, 0.06), _at(Vector3(-0.2, 1.08, -0.42)), e)
			Data.add_box(body, Vector3(0.2, 0.16, 0.06), _at(Vector3(0.2, 1.08, -0.42)), e)
			Data.add_box(body, Vector3(0.3, 0.5, 0.3), _at(Vector3(0, 0.4, 0.1)), b)
			pa = Vector3(-0.4, 1.25, 0.1)
			pb = Vector3(0.4, 1.25, 0.1)
			Data.add_box(la, Vector3(0.7, 0.08, 0.5), _at(Vector3(-0.35, 0, 0)), b)
			Data.add_box(lb, Vector3(0.7, 0.08, 0.5), _at(Vector3(0.35, 0, 0)), b)
		_:
			Data.add_box(body, Vector3(1.0, 0.9, 0.55), _at(Vector3(0, 1.45, 0)), a)
			Data.add_box(body, Vector3(0.55, 0.5, 0.5), _at(Vector3(0, 2.15, 0)), b)
			Data.add_box(body, Vector3(0.34, 0.1, 0.06), _at(Vector3(0, 2.2, -0.27)), e)
			pa = Vector3(0, 1.4, 0)
			pb = Vector3(0, 1.4, 0)
			# 左臂 + 右腿一组，右臂 + 左腿一组
			Data.add_box(la, Vector3(0.34, 1.0, 0.36), _at(Vector3(-0.68, -0.1, 0)), b)
			Data.add_box(la, Vector3(0.38, 1.0, 0.4), _at(Vector3(0.26, -0.9, 0)), a)
			Data.add_box(lb, Vector3(0.34, 1.0, 0.36), _at(Vector3(0.68, -0.1, 0)), b)
			Data.add_box(lb, Vector3(0.38, 1.0, 0.4), _at(Vector3(-0.26, -0.9, 0)), a)
	var out := [body.commit(), la.commit(), lb.commit(), pa, pb]
	_mesh_cache[key] = out
	return out

## 资源库模型：按体型缩到合适大小、脚踩地（浮空的抬到半空），接上动画
func _build_model(key: String) -> bool:
	_model = Models.place(key, _body, Models.shadows)
	if _model.is_empty():
		return false
	var root: Node3D = _model["root"]
	var bb: AABB = _model["aabb"]
	var own_elite := elite and def.has("elite_model")
	# mh：包围盒里真正算身高的比例（翅膀 / 尾巴甩得很高的模型用）；mk：额外放大
	var h := bb.size.y * float(def.get("elite_mh" if own_elite else "mh", 1.0))
	var fit: Array = FIT[str(def["shape"])]
	var k := minf(float(fit[0]) / maxf(h, 0.01), float(fit[1]) / maxf(maxf(bb.size.x, bb.size.z), 0.01))
	k *= float(def.get("elite_mk" if own_elite else "mk", 1.0))
	var lift := maxf(1.0 - h * k * 0.5, 0.0) if def["shape"] == "floater" else 0.0
	root.scale = Vector3.ONE * k
	root.position.y = -bb.position.y * k + lift
	_top = (lift + h * k) * _size
	_mid = (lift + h * k * 0.52) * _size
	_anim = _model["anim"]
	_clips = _model["clips"]
	_overlay = ShaderMaterial.new()
	_overlay.shader = OVERLAY
	_overlay.set_shader_parameter("tint", (def["cols"] as Array)[2])
	_overlay.set_shader_parameter("rim", 0.6 if boss else (0.5 if elite else 0.0))
	_set_overlay(boss or elite)
	return true


func _set_overlay(on: bool) -> void:
	if _overlay == null or on == _overlay_on:
		return
	_overlay_on = on
	for mi: MeshInstance3D in _model["meshes"]:
		mi.material_overlay = _overlay if on else null

## 按当前状态放动画：闲逛走 / 追击跑 / 蓄势到出手放一遍攻击 / 硬直和被打飞放受击；spd = 实际移动速度
func _drive_anim(spd: float) -> void:
	if _anim == null:
		return
	var near := player == null or global_position.distance_squared_to(player.global_position) < ANIM_RANGE * ANIM_RANGE
	if _anim.active != near:
		_anim.active = near
	if not near:
		return
	var role := "idle"
	var rate := 1.0
	var once := false
	match state:
		S.IDLE:
			if spd > 0.5:
				role = "walk"
				rate = clampf(spd / 2.2, 0.6, 1.8)
		S.CHASE:
			if spd > 1.0:
				role = "run"
				rate = clampf(spd / maxf(_spd, 1.0), 0.6, 1.3)
		S.WIND, S.STRIKE, S.SLAM:
			role = "attack"
			once = true
		S.STUN, S.BLOWN:
			role = "hurt"
			once = true
	if not _clips.has(role):
		role = "idle"
		once = false
		rate = 1.0
	if not _clips.has(role):
		return
	var clip := str(_clips[role])
	if once:
		var fresh := _anim_dirty or state != _anim_state
		# 攻击动画横跨蓄势 + 出手两段：从蓄势进到出手不重放
		if fresh and not (role == "attack" and _anim_role == "attack" and state == S.STRIKE):
			var span := 1.0
			if role == "attack":
				span = 1.3 if state == S.SLAM else (1.1 if boss else 0.95)
				span = clampf(_anim.get_animation(clip).length / span, 0.6, 2.6)
			_anim.speed_scale = 1.0
			_anim.play(clip, 0.08, span)
			_anim.seek(0.0, true)
	else:
		if _anim.current_animation != clip:
			_anim.play(clip, 0.2)
		_anim.speed_scale = rate
	_anim_state = state
	_anim_role = role
	_anim_dirty = false


## 死了：模型留在原地把死亡动画放完再缩没（本体照常马上删掉，掉落 / 结算不等它）
func _leave_corpse() -> void:
	if _anim == null or not _clips.has("death") or not is_inside_tree():
		return
	var holder := Node3D.new()
	get_parent().add_child(holder)
	holder.global_transform = global_transform
	_body.rotation.x = 0.0
	_body.reparent(holder, true)
	_set_overlay(false)
	var clip := str(_clips["death"])
	_anim.active = true
	_anim.speed_scale = 1.0
	_anim.play(clip, 0.05)
	_anim.seek(0.0, true)
	var hold := minf(_anim.get_animation(clip).length, 2.0)
	var p := holder.global_position
	var tw := holder.create_tween()
	if not _float:
		tw.tween_property(holder, "global_position:y", minf(p.y, _ground(p.x, p.z)), 0.2)
	tw.tween_interval(hold + 0.4)
	tw.tween_property(holder, "scale", Vector3.ONE * 0.02, 0.35)
	tw.tween_callback(holder.queue_free)


func _build_blocks() -> void:
	var m := StandardMaterial3D.new()
	m.vertex_color_use_as_albedo = true
	m.roughness = 0.85
	m.cull_mode = BaseMaterial3D.CULL_DISABLED
	m.emission_enabled = true
	m.emission = Color.WHITE
	m.emission_energy_multiplier = 0.0
	m.set_meta("glow", false)
	if elite:
		# 精英：整身泛着发光色
		var ec: Color = (def["cols"] as Array)[2]
		m.emission = ec
		m.emission_energy_multiplier = 0.55
		m.set_meta("glow", true)
	_mats.append(m)
	var ms := _meshes(def)
	for i in 3:
		var mi := MeshInstance3D.new()
		mi.mesh = ms[i]
		mi.material_override = m
		mi.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
		if i == 0:
			_body.add_child(mi)
		else:
			var piv := Node3D.new()
			piv.position = ms[2 + i]
			_body.add_child(piv)
			piv.add_child(mi)
			_legs.append(piv)


func top_y() -> float:
	return _top

## 量身体的长宽：有模型的照待机姿势的包围盒（角上是空的，水平方向收一成半）；方块造型按体型给个大概

		var bb: AABB = root.transform * (_model["aabb"] as AABB)

		_half_z = maxf(bb.size.z * 0.5 * 0.85 * _size, 0.3)
		var c := bb.get_center()
		_body_off = Vector2(c.x, c.z) * _size
		return
	match str(def["shape"]):
		"beast":
			_half_x = 0.45 * _size
			_half_z = 0.95 * _size
		"brute":
			_half_x = 0.65 * _size

			_half_x = 0.5 * _size

## 从 from（世界坐标）那个方向看过来，身体表面离脚底原点的水平距离（米）：把身体当成一个按朝向摆的椭圆。
## 主角贴身出手站多远、瞬移落在哪儿（air_melee.gd 的 _gap）、地面挥砍够不够得着（player_melee.gd）、自己追到多近都问它。
## 被打飞翻滚着的时候按最长的那一边算
func body_reach(from: Vector3) -> float:
	var big := maxf(_half_x, _half_z)
	if state == S.BLOWN or state == S.DEAD:
		return big
	var u := Vector2(from.x - global_position.x, from.z - global_position.z)
	if u.length_squared() < 1e-4:
		return big
	u = u.normalized().rotated(_facing)       # 世界 → 本地
	var ix := 1.0 / (_half_x * _half_x)
	var iz := 1.0 / (_half_z * _half_z)
	var a := u.x * u.x * ix + u.y * u.y * iz
	var b := u.x * _body_off.x * ix + u.y * _body_off.y * iz
	var c := _body_off.x * _body_off.x * ix + _body_off.y * _body_off.y * iz - 1.0
	var disc := b * b - a * c
	if disc <= 0.0:
		return big
	return maxf((b + sqrt(disc)) / a, 0.3)


## 不让主角和自己的身体叠在一起（怪没有物理碰撞体，全靠这里兜底）：主角进了「身体表面 + PERSONAL」以内、上下也对得上，就分开。
##   1 是自己这一帧往里拱进去的（追击、扑击冲过头）：把拱进去的那一截退回来 —— 扑击因此停在人身前，不再整个身子压上去；
##   2 剩下的（主角自己往里走 / 本来就叠着）：小怪让开；首领 / 精英沉，让不动，把主角往外推（每帧只挪一部分，不会一下弹开）。
## 主角化作遁光 / 闪现途中是穿过去的，不管。联机木偶的位置听房主的，只推主角。mv = 自己这一帧挪了多少
func _keep_apart(cdt: float, mv: Vector3) -> void:
	if state == S.DEAD or state == S.BLOWN or player == null or player.dead or player.dun_active or player.is_dashing():
		return
	var pp := player.global_position
	if pp.y > global_position.y + _top + 0.2 or pp.y + 1.8 < global_position.y - 0.2:
		return
	var d := Vector3(pp.x - global_position.x, 0.0, pp.z - global_position.z)
	var dl := d.length()
	var need := body_reach(pp) + PERSONAL
	if dl >= need:
		return
	var out := d / dl if dl > 0.05 else Basis(Vector3.UP, _facing) * Vector3.FORWARD
	var over := need - dl
	if not puppet:
		var undo := minf(maxf(mv.dot(out), 0.0), over)
		global_position -= out * undo
		over -= undo
		if undo > 0.0 and _vel.dot(out) > 0.0:
			_vel -= out * _vel.dot(out)        # 往人身上去的那部分速度也收掉（扑击的余势不再接着往里顶）
	if over <= 0.001:
		return
	var step := minf(over, 14.0 * cdt + over * 0.3)
	if boss or elite or puppet:
		# 推主角：脚底贴着地面时直接测会被地面挡住，抬一点再测前面有没有墙
		var xf := player.global_transform
		xf.origin.y += 0.08
		if not player.test_move(xf, out * step):
			player.global_position += out * step
	else:
		global_position -= out * step

func _tick_puppet(dt: float) -> void:
	_net_stale += dt
	if _net_stale > 1.5:
		queue_free()   # 主人不再发它了（走远回收 / 主人离开）
		return
	var before := global_position
	global_position = global_position.lerp(_net_pos, 1.0 - exp(-12.0 * dt)) if global_position.distance_to(_net_pos) < 25.0 else _net_pos
	_facing = lerp_angle(_facing, _net_yaw, 1.0 - exp(-10.0 * dt))
	rotation.y = _facing
	var spd := global_position.distance_to(before) / maxf(dt, 1e-3)
	_phase += dt * (4.0 + spd * 1.2)
	var sw := sin(_phase) * clampf(spd / maxf(_spd, 1.0), 0.0, 1.0)
	for i in _legs.size():
		if _float:
			_legs[i].rotation.z = sin(_phase * 2.0) * 0.6 * (1.0 if i == 0 else -1.0)
		else:
			_legs[i].rotation.x = sw * (0.7 if i % 2 == 0 else -0.7)
	_tick_flash(dt)
	_drive_anim(spd)
	_bar.scale.x = maxf(hp / hp_max, 0.001)


## 区域的轮廓光：暗的区域里、离主角不远的怪都勾上（强度 / 颜色变了才改着色器参数）
func _tick_edge() -> void:
	if _overlay == null:
		return
	var want := 0.0
	if look_rim > 0.02 and (player == null or global_position.distance_squared_to(player.global_position) < EDGE_RANGE * EDGE_RANGE):
		want = look_rim
	if absf(want - _edge) > 0.02 or (want == 0.0 and _edge != 0.0):
		_edge = want
		_overlay.set_shader_parameter("edge", _edge)
		_set_overlay(_flash > 0.0 or boss or elite or _edge > 0.0)
	if _edge > 0.0 and not look_rim_col.is_equal_approx(_edge_col):
		_edge_col = look_rim_col
		_overlay.set_shader_parameter("edge_col", _edge_col)


func _tick_flash(dt: float) -> void:
	_tick_edge()
	if _flash <= 0.0:
		return
	_flash = maxf(_flash - dt * 7.0, 0.0)
	if _overlay != null:
		_set_overlay(_flash > 0.0 or boss or elite or _edge > 0.0)
		_overlay.set_shader_parameter("flash", _flash)
	for m in _mats:
		if not bool(m.get_meta("glow")):
			m.emission_energy_multiplier = _flash * 1.3

## 返回 2 被弹反 / 瞬身 · 1 被挡 · 0 命中 · -1 无效
func _hit_player(dmg: float, heavy: bool, dir: Vector3, ranged: bool) -> int:
	if _target_valid():
		decoy.call("mob_hit", dmg, dir, self, heavy)
		return 0
	if atk_out.is_valid():
		dmg = float(atk_out.call(dmg))
	if player.st == Player.St.GROUND:
		return player.melee.receive_attack(self, {"dmg": dmg, "heavy": heavy, "block_pst": 14.0, "hit_pst": 10.0})
	if player.st == Player.St.AIR:
		return player.air.receive_attack(self, {"dmg": dmg, "heavy": heavy, "kb": 22.0 if heavy else 6.0, "dir": dir,
				"contact": player.center(), "ranged": ranged})
	return -1


func _fire(n: int) -> void:
	var cols: Array = def["cols"]
	var from := center() + Basis(Vector3.UP, _facing) * Vector3(0.0, 0.2, -0.8 * _size)
	var aim: Vector3 = (decoy.call("center") as Vector3) if _target_valid() else player.center() + player.velocity * 0.2
	var to := aim - from
	var d := to.normalized() if to.length_squared() > 0.01 else Vector3.FORWARD
	for i in n:
		var a := (float(i) - float(n - 1) * 0.5) * 0.22
		var mi := Vfx.persistent_sprite(cols[2], 2)
		_bolts.append([mi, from, d.rotated(Vector3.UP, a) * (26.0 + 2.0 * float(region)), 2.4])
	Vfx.cast_flash(from, cols[2], 0.6 * _size)
	Sfx.play("swing", -9.0, 1.4)


func _tick_bolts(dt: float) -> void:
	var i := _bolts.size() - 1
	while i >= 0:
		var b: Array = _bolts[i]
		var pos: Vector3 = b[1]
		var v: Vector3 = b[2]
		var life: float = float(b[3]) - dt
		var np := pos + v * dt
		var done := life <= 0.0
		if not done:
			var c: Vector3 = (decoy.call("center") as Vector3) if _target_valid() else player.center()
			if Geometry3D.get_closest_point_to_segment(c, pos, np).distance_to(c) < 0.9:
				if _hit_player(atk * 0.7, false, v.normalized(), true) != -1:
					Vfx.hit(c, v.normalized(), (def["cols"] as Array)[2], 0.8)
					done = true
		if done:
			(b[0] as Node).queue_free()
			_bolts.remove_at(i)
		else:
			b[1] = np
			b[3] = life
			Vfx.place_streak(b[0] as Node3D, np, v.normalized(), 1.4, 0.2)
		i -= 1


# ---------- 每帧 ----------

func _process(dt: float) -> void:
	var cdt := Combat.cdt(dt)
	_t += cdt
	_cd -= cdt
	_slam_cd -= cdt
	_tick_bolts(dt)
	if puppet:
		_tick_puppet(dt)
		return
	if state == S.DEAD or player == null:
		return
	var lured := _target_valid()
	var pc: Vector3 = (decoy.call("center") as Vector3) if lured else player.center()
	var to := pc - center()
	var dist := to.length() - hit_radius()
	var flat := Vector3(to.x, 0.0, to.z)
	# 身体表面到对方中心的水平距离：追到多近、扑多远都按它（不然身子长的会把人整个盖住）
	var sdist := flat.length() - body_reach(pc)
	# 判中用的：身体表面到对方的距离，带上高度差（只看水平的话，对方在头顶几十米也算「贴身」）
	var sreach := Vector2(maxf(sdist, 0.0), maxf(absf(to.y) - _mid, 0.0)).length()
	# 起手用的：水平上已经站到身前（追击就停在这儿，不再往人脚底下钻），高度差还在扑得到的范围里（和原来按中心算的出手距离同一个数）
	var in_front := sdist < KEEP_SURF + 0.9 and absf(to.y) < 2.6 * _size + 0.8
	var fdir := flat.normalized() if flat.length_squared() > 0.01 else Vector3.FORWARD
	var p := global_position
	var gy := _ground(p.x, p.z)
	var hover := gy + (2.2 if _float else 0.0)
	var cols: Array = def["cols"]

	match state:
		S.IDLE:
			# 刷怪点的怪离点太远（追丢了正往回走）：先回到拴绳六成以内再认人，回去的路上走快一点
			var homing := anchor.is_finite() and Vector2(p.x - anchor.x, p.z - anchor.z).length() > LEASH * (2.0 if boss else 1.0) * 0.6
			_wander_t -= cdt
			if _wander_t <= 0.0:
				_wander_t = randf_range(2.0, 5.0)
				var a := randf() * TAU
				_wander = home + Vector3(cos(a), 0.0, sin(a)) * randf_range(2.0, 9.0)
			var w := _wander - p
			w.y = 0.0
			_vel = w.limit_length(1.0) * _spd * (0.6 if homing else 0.25)
			if w.length() > 0.5:
				_facing = lerp_angle(_facing, atan2(-w.x, -w.z), 1.0 - exp(-4.0 * cdt))
			if not homing and (lured or (dist < AGGRO * (1.6 if boss else 1.0) and not player.dun_active)):
				_enter(S.CHASE)
				Vfx.glint(center() + Vector3.UP * _size, cols[2], 1.0 * _size)
		S.CHASE:
			_facing = lerp_angle(_facing, atan2(-fdir.x, -fdir.z), 1.0 - exp(-8.0 * cdt))
			if roaming:
				home = p
			if p.distance_to(home) > LEASH * (2.0 if boss else 1.0) or dist > 90.0:
				_enter(S.IDLE)
				home = p if not anchor.is_finite() else anchor      # 刷怪点的怪：走回点上
				if anchor.is_finite():
					_wander_t = 0.0
			else:
				var keep := 9.0 if _ranged and not boss else 1.6 * _size
				var want := fdir * _spd * clampf((dist - keep) / 2.0, -0.6, 1.0)
				if not _ranged or boss:
					# 近战的：身体表面停在离对方 KEEP_SURF 米的地方（两个条件取近的那个不再往前拱）
					want = fdir * _spd * clampf(minf(dist - keep, sdist - KEEP_SURF) / 2.0, -0.6, 1.0)
				_vel = _vel.move_toward(want, 40.0 * cdt)
				if _float:
					hover = clampf(pc.y - 0.9, gy + 1.5, gy + 60.0)
				if boss and _slam_cd <= 0.0 and dist < 9.0:
					_slam_cd = randf_range(7.0, 10.0)
					_enter(S.SLAM, 1.0)
					_slam_pos = Vector3(p.x, gy, p.z)
					_vel = Vector3.ZERO
					Vfx.cast_charge(_slam_pos + Vector3.UP * 0.2, Color(1.0, 0.3, 0.2), 1.0)
					Sfx.play("swing_heavy", -6.0, 0.6)
				elif _cd <= 0.0 and ((_ranged and dist < 26.0) or dist < 2.6 * _size + 0.8 or in_front):
					_enter(S.WIND, 0.55 if not boss else 0.7)
					Vfx.glint(center() + fdir * 0.6 * _size, Color(1.0, 0.9, 0.8), 0.9 * _size)
		S.WIND:
			_vel = _vel.move_toward(-fdir * 1.5, 30.0 * cdt)
			_facing = lerp_angle(_facing, atan2(-fdir.x, -fdir.z), 1.0 - exp(-10.0 * cdt))
			if _t >= _dur:
				_enter(S.STRIKE, 0.4)
				_hit_done = false
				var melee := dist < 4.0 * _size + 1.0
				if _ranged and not melee:
					_fire(5 if boss else 1)
					_hit_done = true
				else:
					# 扑到身体表面离对方 0.8 米为止（以前是中心扑到只剩 1 米：首领整个身子压在人身上）
					_vel = fdir * clampf(minf(dist - 1.0, sdist - 0.8), 0.0, 6.0 * _size) / 0.12
					if _float or lured or player.st != Player.St.GROUND:
						_vel.y = clampf(to.y, -6.0, 8.0) / 0.12
					Sfx.play("swing_heavy" if boss else "swing", -7.0, 0.85)
		S.STRIKE:
			if not _hit_done and _t >= 0.12:
				_hit_done = true
				if dist < 2.4 * _size + 0.6 or sreach < 1.7:
					var res := _hit_player(atk * (1.5 if boss else 1.0), boss, to.normalized(), false)
					if res == 2:
						_enter(S.STUN, 1.0)
			if _t >= 0.12:
				_vel = _vel.move_toward(Vector3.ZERO, 90.0 * cdt)
			if _t >= _dur and state == S.STRIKE:
				_enter(S.RECOVER, 0.7 if not boss else 0.9)
				_cd = randf_range(1.2, 2.2)
		S.SLAM:
			if _t >= _dur:
				Vfx.dun_slam(_slam_pos, cols[2], 1.6)
				if not Vfx.quake(_slam_pos, 5.0):
					Vfx.ground_break(_slam_pos, 1.3, 5.0)
				Sfx.play("slam", -1.0, 0.7)
				player.cam.add_trauma(0.5)
				var d := (pc - Vector3.UP) - _slam_pos
				if Vector2(d.x, d.z).length() < 8.0 and absf(d.y) < 4.0:
					_hit_player(atk * 2.0, true, (d.normalized() + Vector3.UP * 0.6).normalized(), false)
				_enter(S.RECOVER, 1.1)
		S.RECOVER:
			_vel = _vel.move_toward(Vector3.ZERO, 30.0 * cdt)
			if _t >= _dur:
				_enter(S.CHASE)
		S.STUN:
			_vel = _vel.move_toward(Vector3.ZERO, maxf(_vel.length() * 5.0, 8.0) * cdt)
			if _t >= _dur:
				_enter(S.CHASE)
		S.BLOWN:
			_vel.y -= G * cdt
			_vel -= _vel * 0.6 * cdt
			if p.y <= gy + 0.05 and _vel.y < 0.0:
				var spd := _vel.length()
				global_position.y = gy
				if spd > 12.0:
					Vfx.dun_slam(Vector3(p.x, gy, p.z), cols[2], 0.9)
					Vfx.crater(Vector3(p.x, gy, p.z), clampf(spd / 34.0, 0.6, 1.2))
					Sfx.play("slam", -4.0, 0.9)
					player.cam.add_trauma(0.3)
					_vel = Vector3.ZERO
					_take_raw(hp_max * 0.12, Vector3.DOWN, true)
					if state == S.DEAD:
						return
				_enter(S.STUN, 0.7)
			elif _t >= _dur:
				_enter(S.STUN, 0.3)

	if not Combat.frozen():
		global_position += _vel * cdt
		_keep_apart(cdt, _vel * cdt)
	if state != S.BLOWN:
		# 贴地 / 悬浮高度
		var ty := hover if state != S.STRIKE or _float else maxf(global_position.y, gy)
		if state == S.STRIKE and not _float:
			_vel.y -= G * cdt
			global_position.y = maxf(global_position.y, gy)
		else:
			global_position.y = lerpf(global_position.y, ty, 1.0 - exp(-8.0 * cdt))
		if global_position.y < gy:
			# 往下扑（主角在脚下 / 倒在地上）的余势别把身子带进地里：以前收招那零点几秒会沉下去大半米
			global_position.y = gy
			_vel.y = maxf(_vel.y, 0.0)
	rotation.y = _facing

	# 简单的肢体摆动：走动时腿前后摆，蓄势时下沉，浮空的上下浮动
	_phase += cdt * (4.0 + _vel.length() * 1.2)
	var sw := sin(_phase) * clampf(_vel.length() / maxf(_spd, 1.0), 0.0, 1.0)
	for i in _legs.size():
		if _float:
			_legs[i].rotation.z = sin(_phase * 2.0) * 0.6 * (1.0 if i == 0 else -1.0)   # 扇翅
		else:
			_legs[i].rotation.x = sw * (0.7 if i % 2 == 0 else -0.7)
	var squash := 0.0
	if state == S.WIND or state == S.SLAM:
		squash = clampf(_t / maxf(_dur, 0.01), 0.0, 1.0) * (0.22 if _anim == null else 0.08)
	_body.scale = Vector3(1.0 + squash * 0.5, 1.0 - squash, 1.0 + squash * 0.5) * _size
	_body.position.y = sin(_phase * 0.5) * 0.12 if _float else 0.0
	if state == S.BLOWN:
		_body.rotation.x += 9.0 * cdt
	else:
		_body.rotation.x = lerp_angle(_body.rotation.x, (-0.35 if _anim == null else -0.1) if state == S.STRIKE else 0.0, 1.0 - exp(-12.0 * cdt))
	_tick_flash(dt)
	_drive_anim(_vel.length())
	if _bar.visible:
		_bar.scale.x = maxf(hp / hp_max, 0.001)
