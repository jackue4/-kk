## 擂台对决（2026-11-23）。用户原话：「擂台支持1v1 pvp格斗，玩家站在上面进行匹配，最多等待10秒，10秒内如果无真人玩家加入，
## 则匹配对应玩家等级的ai陪练，玩家在擂台等待期间如有另外一名玩家加入，则直接进入对决，pvp模式下，要做一点数值平衡，
## 段位差距很大的玩家，需要优化一下平衡，不要数值碾压的太过，主要是让玩家体验pvp格斗的流程」。
## 主城两侧各一座擂台（hub_arena.gd）。这个节点挂在主城（hub_hall.gd）下面，管一场对决从头到尾：
##   · 空闲：人走上台面站稳 STAND_T 秒 → 开始匹配（不用按按钮；走下台就取消）
##   · 匹配（最多 QUEUE_T = 10 秒）：局域网里找同一座擂台上也在排的真人（arena_net.gd）——先听一小会儿，没人就自己开一间等人；
##     有人进来（或听到别人开着的）立刻开打。十秒没等到 → 来一位和自己同境界、同等级的 AI 陪练（air_foe.gd 那套真人式打法）。
##     网页版没有局域网联机，十秒后一定是 AI 陪练
##   · 入场：两人分站台心两侧 MARK 米，倒数 3 2 1（这段时间谁也打不动谁；人定在自己的位置上，这时按「攻」/ 推摇杆不会冲出去）
##   · 对决（最多 FIGHT_T 秒）：各自一条 100 点的气血条，打空的输；到时间看谁剩得多。离台心超过 OUT_R 米算出界，OUT_T 秒内不回来判负
##   · 结算：亮「胜 / 负 / 平」，气血灵力回满，输的人回到台边；想接着打：人在台上挪一步（REARM_T 秒后）就接着匹配下一场——
##     站着不动不会自动再排（挂机的人不会被一场接一场地打）
## 数值（擂台规矩，都在 combat_stats.gd 的「擂台切磋」一节）：不拿攻击力 / 气血的实数对砍——两边同一条 100 点的条，一招只看招式单位；
## 修为高的只多一点（每级 1.2%，封顶 ±12%）；一下最多 14 点、连着打每秒 13 点。境界带来的招式差别（飞不飞得起来、几柄剑）不抹平。
## 联机这边：对 net.gd 来说这个节点就是「大世界」（它只认下面「给 net.gd 用的接口」那几个函数和 lv / _in_trib / level 三个量）。
## 故意不写 class_name、不 preload hub_hall.gd（它 preload 这个文件）：hub 不标类型。

# ---------- 每帧 ----------

func _process(dt: float) -> void:
	if hub == null or player == null or not bool(hub.get("active")):
		return
	match phase:
		Ph.IDLE:
			_tick_idle(dt)
		Ph.QUEUE:
			_tick_queue(dt)
		Ph.READY:
			_tick_ready(dt)
		Ph.FIGHT:
			_tick_fight(dt)
		Ph.RESULT:
			_t += dt
			if _t >= RESULT_T:
				_cleanup()
	_refresh_ui()


## 现在站在哪座擂台的台面上（-1 = 不在 / 现在不能匹配）
func _on_arena() -> int:
	if str(hub.get("area")) != "town" or bool(hub.call("_busy")) or bool(hub.get("meditating")) or player.dead:
		return -1
	if main != null and (main.get("level") != null or bool(main.get("_switching"))):
		return -1
	var p := player.global_position
	if p.y < Arena.H - 0.6 or p.y > Arena.H + 2.2:
		return -1
	return Arena.arena_at(p.x, p.z)


func _tick_idle(dt: float) -> void:
	_rearm = maxf(_rearm - dt, 0.0)
	var s := _on_arena()
	if s < 0:
		_rest = Vector3.INF      # 走下台了：下回上来照常排
	if s < 0 or _rearm > 0.0 or (_rest != Vector3.INF and player.global_position.distance_to(_rest) < REARM_MOVE):
		_stand = 0.0
		return
	_stand += dt
	if _stand >= STAND_T:
		_queue_begin(s)


# ---------- 匹配 ----------

func _queue_begin(s: int) -> void:
	side = s
	phase = Ph.QUEUE
	_t = 0.0
	_stand = 0.0
	_joining = false
	lv = int(Profile.cult(player.realm_idx)[0])
	_host_at = randf_range(0.5, 1.1)
	if _net != null and use_net:
		_net.queue_begin(side, lv)
	Sfx.play("perfect", -12.0, 0.8)
	_log("开始匹配 %s Lv.%d 联机=%s" % [Arena.NAMES[side], lv, _net != null and use_net])


func _queue_cancel(why: String) -> void:
	phase = Ph.IDLE
	_rearm = 1.0
	if _net != null:
		_net.queue_end()
		_net.leave()
	if why != "":
		hub.call("say", why)
	_log("取消匹配：" + why)


func _tick_queue(dt: float) -> void:
	_t += dt
	if _on_arena() != side:
		_queue_cancel("已走下擂台 · 匹配取消")
		return
	if _net != null and use_net and not _joining:
		if _net.active and _net.welcomed and _net.players.size() >= 2:
			_begin_pvp()
			return
		var room := _net.pick_room()
		if not room.is_empty():
			_join(room)
		elif not _net.active and _t >= _host_at:
			if not _net.open_room():
				_host_at = INF      # 端口都被占了：只听，不开房
	if _t >= queue_t and not _joining:
		_begin_ai()


func _join(room: Dictionary) -> void:
	_joining = true
	_log("听到 %s 开着的房间（%s:%d），加入" % [str(room["name"]), str(room["ip"]), int(room["port"])])
	var ok: bool = await _net.join_room(str(room["ip"]), int(room["port"]), 2.5)
	_joining = false
	if phase != Ph.QUEUE:
		if phase == Ph.IDLE:
			_net.leave()      # 等的这一会儿人走下台了
		return
	if ok:
		_begin_pvp()
	else:
		_net.found.clear()
		_host_at = _t + randf_range(0.2, 0.6)

func _mark(west: bool) -> Vector3:
	return center() + Vector3(-MARK if west else MARK, 0.1, 0.0)


func _begin_common(k: String, my_mark: Vector3, their_mark: Vector3) -> void:
	kind = k
	phase = Ph.READY
	_t = 0.0
	_count = 3
	_out = 0.0
	_out_said = 0
	_heat = 0.0
	_heat_out = 0.0
	_rival_dead_t = 0.0
	_rival_lost = false
	_tw = -1
	_foe_dead = false
	last_result = ""
	last_reason = ""
	# 气血灵力回满，站到自己那一侧，面朝对手
	Vfx.vanish(player.center(), Realm.line_color(player.realm))
	player.force_ground()
	player.revive()
	player.global_position = my_mark
	player.velocity = Vector3.ZERO
	_my_mark = my_mark
	var d := their_mark - my_mark
	var yaw := atan2(-d.x, -d.z)
	player.face_now(yaw)
	if cam != null:
		cam.snap(yaw, CameraRig.PITCH_DEFAULT, CameraRig.ZOOM_DEFAULT)
	Vfx.reappear(player.center(), Realm.line_color(player.realm))
	_set_hud(true)
	_pop_big("3", UiStyle.TEXT)
	Sfx.play("perfect", -6.0, 1.0)
	duel_started.emit(k)


func _begin_ai() -> void:
	if _net != null:
		_net.queue_end()
		_net.leave()
	var c := center()
	# 主角站在他现在偏向的那一侧，陪练在对面
	var west := player.global_position.x < c.x
	var fm := _mark(not west) + Vector3(0.0, FOE_HOVER, 0.0)
	_rival_name = "陪练·" + str(ANet.NICKS[randi() % ANet.NICKS.size()])
	_rival_lv = lv
	_rival_realm = player.realm_idx
	foe = AirFoe.new()
	foe.player = player
	foe.display_name = _rival_name
	foe.raid = true              # 主角在地面也照打（炼气 / 筑基没有空战）；倒下不重新凝聚
	foe.pro = true
	foe.hostile = true
	foe.lv = lv
	foe.look_i = player.realm_idx
	foe.style = 2
	foe.qi = AI_QI
	foe.hp_max = CS.arena_ai_hp(player.realm_idx)
	foe.dmg_mul = CS.ARENA_AI_DMG
	foe.skill = clampf(0.4 + 0.07 * float(streak), 0.4, 0.9)      # 连胜越多，陪练越认真
	foe.volley_init = player.realm_idx >= 2                        # 结丹起才会朝主角连发飞剑（对波）
	foe.floor_fn = Callable(hub, "ground")
	foe.ai_enabled = false       # 倒数完才动手
	foe.position = fm + Vector3(0.0, FOE_DROP, 0.0)      # 从上面落到站位上（它没事时自己会飞回 home）
	foe.died.connect(_on_foe_died)
	add_child(foe)
	foe.home = fm
	_warm_fx(fm)
	foe.set("_iframe", READY_T + 0.1)      # 倒数这段时间打不动它
	foe.set("_facing", atan2(fm.x - _mark(west).x, fm.z - _mark(west).z))
	Vfx.reappear(foe.center(), AI_QI)
	_begin_common("ai", _mark(west), _mark(not west))
	_sub_line("十息之内无人应战 · %s 入场" % _rival_name)
	_log("AI 陪练入场：%s Lv.%d 境界 %d 气血 %.0f 水平 %.2f" % [_rival_name, lv, player.realm_idx, foe.hp_max, foe.skill])


## 手机 / 网页那套渲染每种着色器第一次画到才现编（网页上一个零点几秒）。开机预热没管到陪练身上的两样：
## 被打飞时身后的气浪拖尾、倒下时碎成方块的消散——不处理的话，第一场一开打、打倒它的那一下各卡一下。
## 挪到入场倒数这两三秒里，都放在台面底下（镜头里、但被台面挡着看不见）：一小段转圈的拖尾、一粒小方块的消散
## （陪练自己的拖尾每帧会被它按「是不是在突进」重设，借不了）
func _warm_fx(at: Vector3) -> void:
	if Vfx.root == null or not is_instance_valid(Vfx.root):
		return
	var under := Vector3(lerpf(at.x, center().x, 0.5), Arena.H - 1.6, at.z)
	var tr := Trail3D.new()
	tr.width = 0.2
	tr.lifetime = 0.4
	tr.point_fn = func() -> Vector3:
		var a := float(Time.get_ticks_msec()) * 0.02
		return under + Vector3(sin(a) * 0.6, 0.0, cos(a) * 0.6)
	tr.emitting = true
	tr.material_override = MeshLib.trail_material(Color.WHITE, Color(0.8, 0.85, 0.95), 0, 0.9)
	Vfx.root.add_child(tr)
	get_tree().create_timer(1.5).timeout.connect(tr.queue_free)
	var sh := Shatter.new()
	sh.edge = AI_QI
	sh.floor_y = under.y - 0.3
	Vfx.root.add_child(sh)
	sh.add_cube(Transform3D(Basis.IDENTITY, under), 0.05, AI_QI)
	sh.start(under, Vector3.DOWN, 0.2)


func _begin_pvp() -> void:
	_net.queue_end()
	_net.enter_world(self)
	var rid := _net.rival_id()
	var info: Dictionary = _net.players.get(rid, {})
	_rival_name = str(info.get("name", "道友"))
	_rival_realm = int(info.get("realm", 0))
	_rival_lv = _rival_realm * 3 + 1      # 先按境界估，对方的状态包到了再换成真的（_refresh_rival）
	# 开房的站西边，进来的站东边（两台机器上算出来是同一个站位）
	_begin_common("pvp", _mark(_net.is_host), _mark(not _net.is_host))
	_sub_line("道友 %s 应战" % _rival_name)
	_log("真人对手：%s（编号 %d，%s）" % [_rival_name, rid, "我开的房" if _net.is_host else "我加入的"])


func _tick_ready(dt: float) -> void:
	_t += dt
	_refresh_rival()
	# 倒数时人定在自己的位置上：这时按了「攻」（结丹起会腾空朝对手冲过去）/ 推了摇杆，都按回原地
	if player.st != Player.St.GROUND:
		_log("入场倒数时主角离了地（状态 %d，位置 %s）：按回原位" % [int(player.st), str(player.global_position.snapped(Vector3.ONE * 0.1))])
		player.force_ground()
	var q := player.global_position
	if Vector2(q.x - _my_mark.x, q.z - _my_mark.z).length() > 0.02 or q.y > _my_mark.y + 0.5:
		player.global_position = Vector3(_my_mark.x, minf(q.y, _my_mark.y + 0.5), _my_mark.z)
	player.velocity = Vector3(0.0, minf(player.velocity.y, 0.0), 0.0)
	if kind == "pvp" and not _rival_here():
		_finish("win", "对手离场")
		return
	var left := int(ceilf((READY_T - _t) / (READY_T / 3.0)))
	if left < _count and left >= 1:
		_count = left
		_pop_big(str(left), UiStyle.TEXT)
		Sfx.play("perfect", -8.0, 1.0 + 0.1 * float(3 - left))
	if _t >= READY_T:
		phase = Ph.FIGHT
		_t = 0.0
		_pop_big("战", UiStyle.GOLD, 0.9)
		Sfx.play("perfect", -2.0, 0.7)
		if is_instance_valid(foe):
			foe.ai_enabled = true
		_log("开打")


# ---------- 对决 ----------

func _rival_here() -> bool:
	return _net != null and _net.active and _net.players.size() >= 2


func _rival_puppet() -> Node3D:
	if _net == null:
		return null
	var r: Variant = _net.remotes.get(_net.rival_id())
	return r as Node3D if is_instance_valid(r) else null


func _refresh_rival() -> void:
	if kind != "pvp":
		return
	var r := _rival_puppet()
	if r != null and bool(r.get("_got")):
		_rival_lv = int(r.get("lv"))
		_rival_realm = int(r.get("realm_idx"))


## 对手的气血（0 ~ 1）
func rival_ratio() -> float:
	if kind == "ai":
		if _foe_dead or not is_instance_valid(foe):
			return 0.0
		return clampf(foe.hp / maxf(foe.hp_max, 1.0), 0.0, 1.0)
	var r := _rival_puppet()
	if r == null or not bool(r.get("_got")):
		return 1.0
	return clampf(float(r.get("hp")) / 100.0, 0.0, 1.0)


func my_ratio() -> float:
	return clampf(player.melee.hp / 100.0, 0.0, 1.0)


func _tick_fight(dt: float) -> void:
	_t += dt
	_heat = maxf(_heat - CS.ARENA_SUSTAIN * dt, 0.0)
	_heat_out = maxf(_heat_out - CS.ARENA_SUSTAIN * dt, 0.0)
	_refresh_rival()
	# 自己倒了（陨落事件没接到时的兜底）
	if player.melee.hp <= 0.0 or player.dead:
		_i_lost("")
		return
	if kind == "ai":
		if _foe_dead or not is_instance_valid(foe):
			_finish("win", "")
			return
	else:
		if _rival_lost:
			_finish("win", last_reason)
			return
		if not _rival_here():
			_finish("win", "对手离场")
			return
		var r := _rival_puppet()
		if r != null and bool(r.get("_got")) and float(r.get("hp")) <= 0.0:
			_rival_dead_t += dt
			if _rival_dead_t > 0.8:
				_finish("win", "")      # 他那边的认负没送到：按同步过来的气血判
				return
		else:
			_rival_dead_t = 0.0
	# 出界
	var c := center()
	var p := player.global_position
	if Vector2(p.x - c.x, p.z - c.z).length() > OUT_R or p.y < -8.0:
		_out += dt
		var left := int(ceilf(OUT_T - _out))
		if left != _out_said and left >= 1:
			_out_said = left
			hub.call("say", "已出擂台 · %d 息内回来" % left, 0.9)
		if _out >= OUT_T:
			_i_lost("出界")
			return
	else:
		_out = 0.0
		_out_said = 0
	# 到时间：看谁剩得多。联机里由开房的一方判（两边各看各的会判出两个结果），另一方等他的话，等不到再自己判
	if _t >= FIGHT_T:
		if kind == "ai" or _net.is_host:
			var mine := my_ratio()
			var theirs := rival_ratio()
			var res := "draw" if absf(mine - theirs) < 0.005 else ("win" if mine > theirs else "lose")
			if kind == "pvp":
				_say_rival("tw", str(0 if res == "draw" else (_net.my_id if res == "win" else _net.rival_id())))
			_finish(res, "时辰到")
		elif _tw >= 0:
			_finish("draw" if _tw == 0 else ("win" if _tw == _net.my_id else "lose"), "时辰到")
		elif _t >= FIGHT_T + 3.0:
			var m2 := my_ratio()
			var t2 := rival_ratio()
			_finish("draw" if absf(m2 - t2) < 0.005 else ("win" if m2 > t2 else "lose"), "时辰到")


func _i_lost(reason: String) -> void:
	if kind == "pvp":
		_say_rival("lose", reason)
	_finish("lose", reason)

func _finish(res: String, reason: String) -> void:
	if phase == Ph.RESULT or phase == Ph.IDLE:
		return
	var took := _t if phase == Ph.FIGHT else 0.0
	phase = Ph.RESULT
	_t = 0.0
	last_result = res
	last_reason = reason
	if res == "win":
		wins += 1
		streak += 1
	elif res == "lose":
		losses += 1
		streak = 0
	if is_instance_valid(foe):
		foe.ai_enabled = false
		if res != "win":
			foe.call("stand_down")
	var line := ""
	match res:
		"win":
			line = "胜 %s" % _rival_name
			if reason != "":
				line += " · " + reason
			if streak >= 2:
				line += " · %d 连胜" % streak
			_pop_big("胜", UiStyle.GOLD, RESULT_T)
			Sfx.play("perfect", -1.0, 1.2)
		"lose":
			line = "惜败于 %s" % _rival_name
			if reason != "":
				line += " · " + reason
			_pop_big("负", Color(0.92, 0.36, 0.3), RESULT_T)
		_:
			line = "与 %s 战平 · %s" % [_rival_name, reason]
			_pop_big("平", UiStyle.TEXT, RESULT_T)
	_sub_line(line, RESULT_T)
	_log("结束：%s %s（我 %.0f%% / 对手 %.0f%%，用时 %.1f 秒）" % [res, reason, my_ratio() * 100.0, rival_ratio() * 100.0, took])
	duel_ended.emit(res)


## 收场：对手退场、断开联机、界面还原、气血灵力回满；输的人（倒下 / 出界）回到台边
func _cleanup() -> void:
	var lost := last_result == "lose"
	if is_instance_valid(foe):
		if not _foe_dead:
			Vfx.vanish(foe.center(), AI_QI)
		foe.queue_free()
	foe = null
	if _net != null:
		_net.queue_end()
		_net.leave()
	_set_hud(false)
	if bool(hub.get("active")):
		if lost or player.dead or player.melee.hp <= 0.0:
			player.force_ground()
			player.revive()
			var c := center()
			var toward := -1.0 if side == 1 else 1.0       # 朝中心广场的那一侧（红毯石阶口）
			player.global_position = c + Vector3(toward * (Arena.R - 7.0), 0.1, 0.0)
			player.velocity = Vector3.ZERO
			player.face_now(atan2(toward, 0.0))
			if cam != null:
				cam.snap(atan2(toward, 0.0), CameraRig.PITCH_DEFAULT, CameraRig.ZOOM_DEFAULT)
			Vfx.reappear(player.center(), Realm.line_color(player.realm))
		else:
			player.revive()
	phase = Ph.IDLE
	kind = ""
	_stand = 0.0
	_rearm = REARM_T
	_rest = player.global_position
	_big.visible = false
	_sub.visible = false
	if _on_arena() >= 0:
		hub.call("say", "挪步再战一场 · 走下擂台就此别过", 2.6)


## 主城关掉了（进关卡 / 去试剑台）：有什么都立刻收掉，不结算
func abort() -> void:
	if phase == Ph.IDLE:
		return
	if phase == Ph.QUEUE:
		_queue_cancel("")
		return
	if kind == "pvp" and phase != Ph.RESULT:
		_say_rival("lose", "对手离场")
	if is_instance_valid(foe):
		foe.queue_free()
	foe = null
	if _net != null:
		_net.queue_end()
		_net.leave()
	_set_hud(false)
	phase = Ph.IDLE
	kind = ""
	_rearm = 1.0
	_big.visible = false
	_sub.visible = false

## 借「全屏播报」那条可靠通道给对手带一句话（标题 §duel）：lose|原因（我认负）/ tw|编号（到时间，房主判谁赢，0 = 平）
func _say_rival(op: String, arg: String) -> void:
	if _net != null and _net.active:
		_net.send_announce("§duel", "%s|%s" % [op, arg])


func net_announce(title_text: String, line: String) -> void:
	if title_text != "§duel" or kind != "pvp":
		return
	var parts := line.split("|")
	if parts.size() < 2:
		return
	if parts[0] == "lose":
		_rival_lost = true
		last_reason = str(parts[1]) if phase == Ph.FIGHT or phase == Ph.READY else last_reason
		if phase == Ph.READY:
			_finish("win", str(parts[1]))
	elif parts[0] == "tw":
		_tw = int(parts[1])


## 我打中了对手的木偶：按擂台规矩折成他气血条上的点数再发给他
func net_deal_hit(puppet: Node3D, peer: int, info: Dictionary) -> void:
	if phase != Ph.FIGHT or _net == null:
		return
	var want := CS.arena_hit(float(info.get("dmg", 10.0)), player.realm_idx, lv, int(puppet.get("lv")))
	var d := CS.arena_clamp(want, _heat_out)
	_heat_out += d
	if d <= 0.0:
		return
	var i := info.duplicate()
	i["dmg"] = d
	_net.send_hit(peer, i)
	Vfx.damage_number(puppet.call("center") + Vector3(0.0, 0.9, 0.0), d, Color(1.0, 0.85, 0.5) if bool(info.get("heavy", false)) else Color.WHITE)


## 对手打中了我（dmg 已经是气血条上的点数）：这边再封一次顶，然后按我现在的状态走正常的受击结算（能挡、能闪）
func net_take_hit(src: Variant, _from: int, dmg: float, dir: Vector3, kb: float, heavy: bool, launch: String, ranged: bool) -> void:
	if phase != Ph.FIGHT or player.dead:
		return
	if player.st == Player.St.AIR and player.air.warned_by(src):
		return      # 对波预警窗口里：他的飞剑先不算
	var d := CS.arena_clamp(dmg, _heat)
	_heat += d
	if d <= 0.0:
		return
	var s: Node3D = src if src != null and is_instance_valid(src) else player
	if player.st == Player.St.GROUND:
		player.melee.receive_attack(s, {"dmg": d, "heavy": heavy, "block_pst": 14.0, "hit_pst": 10.0, "dir": dir})
	elif player.st == Player.St.AIR:
		player.air.receive_attack(s, {"dmg": d, "heavy": heavy or launch != "", "kb": kb, "dir": dir,
				"contact": player.center(), "ranged": ranged, "launch": launch})
	else:
		player.melee.hp = maxf(player.melee.hp - d, 0.0)
		Vfx.hit(player.center(), dir, Color(1.0, 0.5, 0.4), 1.0)
		if player.melee.hp <= 0.0 and not player.dead:
			player.force_ground()
			player.melee.receive_attack(s, {"dmg": 0.0, "heavy": true, "block_pst": 0.0, "hit_pst": 0.0, "dir": dir})

func _pop_big(text: String, col: Color, hold := 0.7) -> void:

	_big.add_theme_color_override("font_color", col)
	_big.visible = true
	_big.modulate.a = 1.0
	_big.pivot_offset = _big.size * 0.5
	_big.scale = Vector2(1.5, 1.5)
	var tw := _big.create_tween()
	tw.tween_property(_big, "scale", Vector2.ONE, 0.16).set_trans(Tween.TRANS_BACK).set_ease(Tween.EASE_OUT)
	tw.tween_interval(maxf(hold - 0.4, 0.05))
	tw.tween_property(_big, "modulate:a", 0.0, 0.24)
